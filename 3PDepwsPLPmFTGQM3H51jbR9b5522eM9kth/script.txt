func dataTotalLocked (inShareAmount,outBaseAmount,inBaseAmount,outShareAmount) = makeString(["%d%d%d%d", toString(inShareAmount), toString(outBaseAmount), toString(inBaseAmount), toString(outShareAmount)], SEP)


func dataTotalLockedInt (inShareAmount,outBaseAmount,inBaseAmount,outShareAmount) = [-1, inShareAmount, outBaseAmount, inBaseAmount, outShareAmount]


func readTotalLocked (key) = {
    let totalLockedArray = split(valueOrElse(getString(this, key), dataTotalLocked(0, 0, 0, 0)), SEP)
    dataTotalLockedInt(parseIntValue(totalLockedArray[IdxTotalLockedInShare]), parseIntValue(totalLockedArray[IdxTotalLockedOutBase]), parseIntValue(totalLockedArray[IdxTotalLockedInBase]), parseIntValue(totalLockedArray[IdxTotalLockedOutShare]))
    }


func calcTotalLockedDiff (direction,operationType,internalBaseAssetStr,price,priceMult,inAmount,baseAssetId,shareAssetId) = {
    let t = (direction + operationType)
    if ((t == "submitP"))
        then {
            let totalDiff = dataTotalLockedInt(0, 0, inAmount, 0)
            let userDiff = totalDiff
            $Tuple4(totalDiff, userDiff, 0, fromBase58String(""))
            }
        else if ((t == "submitG"))
            then {
                let totalDiff = dataTotalLockedInt(inAmount, 0, 0, 0)
                let userDiff = totalDiff
                $Tuple4(totalDiff, userDiff, 0, fromBase58String(""))
                }
            else if ((t == "executeP"))
                then {
                    let outAmount = convertBase2Share(inAmount, price, priceMult)
                    let totalDiff = dataTotalLockedInt(0, 0, 0, outAmount)
                    let userDiff = dataTotalLockedInt(0, 0, inAmount, 0)
                    $Tuple4(totalDiff, userDiff, outAmount, shareAssetId)
                    }
                else if ((t == "executeG"))
                    then {
                        let outAmount = convertShare2Base(inAmount, price, priceMult)
                        let totalDiff = dataTotalLockedInt(0, outAmount, 0, 0)
                        let userDiff = dataTotalLockedInt(inAmount, 0, 0, 0)
                        $Tuple4(totalDiff, userDiff, outAmount, baseAssetId)
                        }
                    else if ((t == "topup"))
                        then {
                            let totalLockedArray = readTotalLocked(keyTotalLocked(internalBaseAssetStr))
                            let totalLockedInBaseAmount = totalLockedArray[IdxTotalLockedInBase]
                            let totalLockedInShareAmount = totalLockedArray[IdxTotalLockedInShare]
                            let totalDiff = dataTotalLockedInt(totalLockedInShareAmount, (-1 * convertShare2Base(totalLockedInShareAmount, price, priceMult)), totalLockedInBaseAmount, (-1 * convertBase2Share(totalLockedInBaseAmount, price, priceMult)))
                            $Tuple4(totalDiff, nil, 0, fromBase58String(""))
                            }
                        else throw(("Unsupported Type " + t))
    }


func TotalLockedStringEntry (action,key,diff) = {
    func UPDATE (a,b) =     if ((action == "INCREMENT"))
        then (a + b)
        else if ((action == "DECREMENT"))
            then (a - b)
            else throw(("Unsupported action " + action))

    let dataArray = readTotalLocked(key)
    StrE(key, dataTotalLocked(UPDATE(dataArray[IdxTotalLockedInShare], diff[IdxTotalLockedInShare]), UPDATE(dataArray[IdxTotalLockedOutBase], diff[IdxTotalLockedOutBase]), UPDATE(dataArray[IdxTotalLockedInBase], diff[IdxTotalLockedInBase]), UPDATE(dataArray[IdxTotalLockedOutShare], diff[IdxTotalLockedOutShare])))
    }

func genericCalcPrice (internalBaseAssetStr,baseAssetId,topUpBaseAmount,shareAssetId,decimalsMultPrice) = {
    let totalLockedArray = readTotalLocked(keyTotalLocked(internalBaseAssetStr))
    let totalLockedOutBaseAmount = totalLockedArray[IdxTotalLockedOutBase]
    let currIterTotalInBaseAmount = totalLockedArray[IdxTotalLockedInBase]
    let baseAssetBalance = assetBalance(this, baseAssetId)
    let baseAssetBalanceWCO = (((baseAssetBalance + topUpBaseAmount) - currIterTotalInBaseAmount) - totalLockedOutBaseAmount)
    let totalLockedOutShareAmount = totalLockedArray[IdxTotalLockedOutShare]
    let currIterTotalInShareAmount = totalLockedArray[IdxTotalLockedInShare]
    let shareEmission = value(assetInfo(shareAssetId)).quantity
    if ((0 > baseAssetBalanceWCO))
        then throw(((("baseAssetBalanceWco < 0: baseAssettBalance=" + toString(baseAssetBalance)) + " baseAssetBalanceWco=") + toString(baseAssetBalanceWCO)))
        else {
            let lastPrice = getIntOrFail(keyPriceLast(internalBaseAssetStr))
            let price = if ((shareEmission == 0))
                then lastPrice
                else fraction(baseAssetBalanceWCO, decimalsMultPrice, shareEmission)
            $Tuple9(price, baseAssetBalance, -1, baseAssetBalanceWCO, shareEmission, currIterTotalInBaseAmount, currIterTotalInShareAmount, totalLockedOutBaseAmount, totalLockedOutShareAmount)
            }
    }


func calcPrice (internalBaseAssetStr,baseAssetId,shareAssetId,decimalsMultPrice) = genericCalcPrice(internalBaseAssetStr, baseAssetId, 0, shareAssetId, decimalsMultPrice)

    case s: String => 
        addressFromStringValue(s)
    case _ =>

}


func managerPublicKeyOrUnit () = {
    let managerVaultAddress = getManagerVaultAddressOrThis()
    match getString(managerVaultAddress, keyManagerPublicKey()) {
        case s: String => 
            fromBase58String(s)
        case _: Unit => 
            unit
        case _ => 
            throw("Match error")
    }
    }


func isManager (i) = match managerPublicKeyOrUnit() {
    case pk: ByteVector => 
        (i.callerPublicKey == pk)
    case _: Unit => 
        (i.caller == this)
    case _ => 
        throw("Match error")
}


func mustManager (i) = if (isManager(i))
    then true
    else throw("permission denied")


func commonSubmit (operationType,i,inAmount,inAssetId,baseAssetStr) = {
    let inAssetStr = toBase58String(inAssetId)
    let userAddressStr = toString(i.caller)
    let baseAssetId = fromBase58String(baseAssetStr)
    let cfgArray = readAssetCfgOrFail(baseAssetStr)
    let shareAssetStr = cfgArray[IdxCfgShareAssetId]
    let shareAssetId = fromBase58String(shareAssetStr)
    let decimalsMultBothAssets = parseIntValue(cfgArray[IdxCfgDecimalsMultBothAssets])
    let internalBaseAssetStr = cfgArray[IdxCfgInternalBaseAsset]
    let limitsKEY = keyLimitsRemaining(internalBaseAssetStr)
    let limitsCfgArray = split(getStringOrFail(limitsKEY), SEP)
    let limitsRemainingBase = parseIntValue(limitsCfgArray[IdxLimitsRemainingBase])
    let limitsRemainingShare = parseIntValue(limitsCfgArray[IdxLimitsRemainingShare])
    let isSubmitBlocked = valueOrElse(getBoolean(this, keyShutdownSubmitOperation(internalBaseAssetStr)), false)
    if (isSubmitBlocked)
        then throw("submit operation is blocked")
        else {
            let operationsMutex = valueOrElse(getInteger(this, keyTopupMutext(internalBaseAssetStr)), 0)
            if (((operationsMutex + 60) > height))
                then throw("submit operations are blocked by topup manager")
                else {
                    let diffTuple = calcTotalLockedDiff("submit", operationType, internalBaseAssetStr, 0, 0, inAmount, baseAssetId, shareAssetId)
                    let limitsRemainingBaseNew = (limitsRemainingBase - diffTuple._2[IdxTotalLockedInBase])
                    let limitsRemainingShareNew = (limitsRemainingShare - diffTuple._2[IdxTotalLockedInShare])
                    if (if ((0 > limitsRemainingBaseNew))
                        then true
                        else (0 > limitsRemainingShareNew))
                        then failSubmitLimitsExceeds(limitsRemainingBase, limitsRemainingShare, limitsRemainingBaseNew, limitsRemainingShareNew)
                        else {
                            let topUpCurrentIdx = getIntOrFail(keyTopUpCurrentIdx(internalBaseAssetStr))
                            ((([StrE(keyOperation(operationType, internalBaseAssetStr, userAddressStr, toBase58String(i.transactionId)), dataOperation("PENDING", inAmount, 0, 0, height, lastBlock.timestamp, 0, 0, (topUpCurrentIdx + 1)))] :+ TotalLockedStringEntry("INCREMENT", keyTotalLocked(internalBaseAssetStr), diffTuple._1)) :+ TotalLockedStringEntry("INCREMENT", keyTotalLockedByUser(internalBaseAssetStr, userAddressStr), diffTuple._2)) :+ RemainingLimitsStringEntry(limitsKEY, limitsRemainingBaseNew, limitsRemainingShareNew))
                            }
                    }
            }
    }


func commonExecute (operationType,baseAssetStr,userAddressStr,submitTxIdStr) = {
    let userAddress = addressFromStringValue(userAddressStr)
    let assetCfgArray = readAssetCfgOrFail(baseAssetStr)
    let shareAssetId = fromBase58String(assetCfgArray[IdxCfgShareAssetId])
    let internalBaseAssetStr = assetCfgArray[IdxCfgInternalBaseAsset]
    let decimalsMultPrice = parseIntValue(assetCfgArray[IdxCfgDecimalsMultPrice])
    let baseAssetId = fromBase58String(baseAssetStr)
    let opKey = keyOperation(operationType, internalBaseAssetStr, userAddressStr, submitTxIdStr)
    let opArray = split(getStringOrFail(opKey), SEP)
    let status = opArray[IdxOperStatus]
    let inAmount = parseIntValue(opArray[IdxOperInAmount])
    let topupUnlockIdx = parseIntValue(opArray[IdxOperTopupUnlockIdx])
    let currTopUpIdx = getIntOrFail(keyTopUpCurrentIdx(internalBaseAssetStr))
    let priceByTopUpId = getIntOrFail(keyPriceByTopUpIdx(internalBaseAssetStr, topupUnlockIdx))
    if ((status != "PENDING"))
        then failExecuteGet("Status is not PENDING", baseAssetStr, userAddressStr, submitTxIdStr, operationType)
        else if ((topupUnlockIdx > currTopUpIdx))
            then failExecuteGet(((("OperLock[" + toString(topupUnlockIdx)) + "] > ") + toString(currTopUpIdx)), baseAssetStr, userAddressStr, submitTxIdStr, operationType)
            else {
                let diffTuple = calcTotalLockedDiff("execute", operationType, internalBaseAssetStr, priceByTopUpId, decimalsMultPrice, inAmount, baseAssetId, shareAssetId)
                let outAmount = diffTuple._3
                let outTransferData = if ((diffTuple._4 == baseAssetId))
                    then [ScriptTransfer(userAddress, outAmount, baseAssetId)]
                    else [ScriptTransfer(userAddress, outAmount, shareAssetId)]
                (((outTransferData :+ StrE(opKey, dataOperationExecutionUpdate(opArray, "FINISHED", priceByTopUpId, outAmount))) :+ TotalLockedStringEntry("DECREMENT", keyTotalLocked(internalBaseAssetStr), diffTuple._1)) :+ TotalLockedStringEntry("DECREMENT", keyTotalLockedByUser(internalBaseAssetStr, userAddressStr), diffTuple._2))
                }
    }

@Callable(i)
func adminRegisterAsset (baseAssetStr,shareAssetName,shareAssetDescr,getDelayinBlocks,shutdownManagerAddress,startPrice,topupIntervalInBlocks,topupMaxNegativePart,topupManagerAddress,submitLimitsBaseMax,submitLimitsBaseReset,submitLimitsShareMax,submitLimitsShareReset,adminAddress) = {
    let baseAssetId = fromBase58String(baseAssetStr)
    let bothAssetsDecimals = value(assetInfo(baseAssetId)).decimals
    let decimalsMultBothAssets = pow(10, 0, bothAssetsDecimals, 0, 0, DOWN)
    let decimalsMultPrice = ((100 * 1000) * 1000)
    let topupMaxNegativePercents = fraction(topupMaxNegativePart, 100, decimalsMultBothAssets)
    let baseAssetBalance = assetBalance(this, baseAssetId)
    let checkCaller = mustManager(i)
    if ((checkCaller == checkCaller))
        then if ((baseAssetBalance == 0))
            then throw(((toString(this) + " must have any initial balance of ") + baseAssetStr))
            else if (isDefined(getString(this, keyAssetCfg(baseAssetStr))))
                then throw((baseAssetStr + " has been already registered"))
                else if ((toString(addressFromStringValue(shutdownManagerAddress)) != shutdownManagerAddress))
                    then throw("invalid shutdownManagerAddress")
                    else if ((toString(addressFromStringValue(topupManagerAddress)) != topupManagerAddress))
                        then throw("invalid topupManagerAddress")
                        else if ((0 > getDelayinBlocks))
                            then throw(("invalid getDelayinBlocks=" + toString(getDelayinBlocks)))
                            else if (if ((0 >= topupMaxNegativePercents))
                                then true
                                else (topupMaxNegativePercents >= 99))
                                then throw("invalid topupMaxNegativePart parameter")
                                else {
                                    let shareInitAmount = convertBase2Share(baseAssetBalance, startPrice, decimalsMultPrice)
                                    let shareAssetIssueAction = Issue(shareAssetName, shareAssetDescr, shareInitAmount, bothAssetsDecimals, true)
                                    let shareAssetId = calculateAssetId(shareAssetIssueAction)
                                    let shareAssetStr = toBase58String(shareAssetId)
                                    let internalBaseAssetId = valueOrElse(getInteger(this, keyNextInternalAssetId()), 0)
                                    let internalBaseAssetStr = toString(internalBaseAssetId)
[StrE(keyAssetCfg(baseAssetStr), dataAssetCfg(shareAssetStr, internalBaseAssetStr, decimalsMultBothAssets, decimalsMultPrice, getDelayinBlocks, topupIntervalInBlocks, topupMaxNegativePart, topupManagerAddress, submitLimitsBaseMax, submitLimitsBaseReset, submitLimitsShareMax, submitLimitsShareReset, adminAddress)), StrE(keyMappingsInternal2baseAssetId(internalBaseAssetId), baseAssetStr), StrE(keyMappingsBaseAsset2internalId(baseAssetStr), internalBaseAssetStr), StrE(keyMappingsShare2baseAssetId(shareAssetStr), baseAssetStr), StrE(keyMappingsBaseAsset2shareId(baseAssetStr), shareAssetStr), BooleanEntry(keyShutdownSubmitOperation(internalBaseAssetStr), false), StrE(keyShutdownManager(internalBaseAssetStr), shutdownManagerAddress), IntE(keyNextInternalAssetId(), (internalBaseAssetId + 1)), IntE(keyPriceLast(internalBaseAssetStr), startPrice), IntE(keyPriceHistory(internalBaseAssetStr, height, lastBlock.timestamp), startPrice), IntE(keyTopUpCurrentIdx(internalBaseAssetStr), 0), RemainingLimitsStringEntry(keyLimitsRemaining(internalBaseAssetStr), submitLimitsBaseMax, submitLimitsShareMax), shareAssetIssueAction, ScriptTransfer(addressFromStringValue(topupManagerAddress), shareInitAmount, shareAssetId)]
                                    }
        else throw("Strict value is not equal to itself.")
    }



@Callable(i)
func shutdownSubmits (internalBaseAssetId) = {
    let checkCaller = mustManager(i)
    if ((checkCaller == checkCaller))
        then {
            let internalBaseAssetIdStr = toString(internalBaseAssetId)
            let baseAssetIdStr = getStringOrFail(keyMappingsInternal2baseAssetId(internalBaseAssetId))
            let shutdownManagerAddress = getStringOrFail(keyShutdownManager(internalBaseAssetIdStr))
            if ((1 > size(baseAssetIdStr)))
                then throw("invalid internalBaseAssetId")
                else [BooleanEntry(keyShutdownSubmitOperation(toString(internalBaseAssetId)), true)]
            }
        else throw("Strict value is not equal to itself.")
    }

@Callable(i)
func topUpBalance (baseAssetStr,income) = {
    let baseAssetId = fromBase58String(baseAssetStr)
    let cfg = readAssetCfgOrFail(baseAssetStr)
    let shareAssetId = fromBase58String(cfg[IdxCfgShareAssetId])
    let priceMult = parseIntValue(cfg[IdxCfgDecimalsMultPrice])
    let bothAssetMult = parseIntValue(cfg[IdxCfgDecimalsMultBothAssets])
    let topupIntervalInBlocks = parseIntValue(cfg[IdxCfgTopupIntervalInBlocks])
    let topupMaxNegativePart = parseIntValue(cfg[IdxCfgTopupMaxNegativePart])
    let internalBaseAssetStr = cfg[IdxCfgInternalBaseAsset]
    let topUpManagerAddressStr = cfg[IdxCfgTopupManagerAddress]
    let submitLimitsBaseMax = parseIntValue(cfg[IdxCfgSubmitLimitsBaseMax])
    let submitLimitsShareMax = parseIntValue(cfg[IdxCfgSubmitLimitsShareMax])
    let topUpCurrentIdxKEY = keyTopUpCurrentIdx(internalBaseAssetStr)
    let prevTopUpIdx = getIntOrFail(topUpCurrentIdxKEY)
    let currentTopUpIdx = (prevTopUpIdx + 1)
    let valid = if ((income > 0))
        then {
            let pmt = value(i.payments[0])
            let pmtAssetId = value(pmt.assetId)
            if ((baseAssetId != pmtAssetId))
                then throw("attached payment's asset id is NOT matched passed baseAssetStr")
                else if ((size(i.payments) > 1))
                    then throw("only one payment can be attached")
                    else if ((pmt.amount != income))
                        then throw("attached payment.amount is NOT matched passed income argument")
                        else true
            }
        else if ((0 > income))
            then {
                let baseBalance = assetBalance(this, baseAssetId)
                let allowedAmount = fraction(topupMaxNegativePart, baseBalance, bothAssetMult)
                if ((-(income) > allowedAmount))
                    then throw(("topup negative income couldn't be greater than " + toString(allowedAmount)))
                    else true
                }
            else throw("zero income is not allowed")
    let topUpLastHeightKEY = keyTopUpLastHeight(internalBaseAssetStr, toString(i.caller))
    let topUpLastHeight = valueOrElse(getInteger(this, topUpLastHeightKEY), 0)
    if ((toString(i.caller) != topUpManagerAddressStr))
        then failTopupManagerOnly(topUpManagerAddressStr)
        else if (!(valid))
            then throw("validation failed")
            else if ((topupIntervalInBlocks > (height - topUpLastHeight)))
                then throw((("1 topup per " + toString(topupIntervalInBlocks)) + " blocks from the same address is allowed"))
                else {
                    let price = genericCalcPrice(internalBaseAssetStr, baseAssetId, income, shareAssetId, priceMult)._1
                    let diffTuple = calcTotalLockedDiff("topup", "", internalBaseAssetStr, price, priceMult, 0, baseAssetId, shareAssetId)
                    let topupTotalDiff = diffTuple._1
                    let priceAthKEY = keyPriceATH(internalBaseAssetStr)
                    let prevPriceATH = valueOrElse(getInteger(this, priceAthKEY), 0)
                    ((((((([IntE(keyPriceLast(internalBaseAssetStr), price), IntE(keyPriceHistory(internalBaseAssetStr, height, lastBlock.timestamp), price), IntE(keyPriceByTopUpIdx(internalBaseAssetStr, currentTopUpIdx), price), IntE(topUpCurrentIdxKEY, currentTopUpIdx), IntE(priceAthKEY, if ((price > prevPriceATH))
                        then price
                        else prevPriceATH), IntE(topUpLastHeightKEY, height)] :+ TotalLockedStringEntry("DECREMENT", keyTotalLocked(internalBaseAssetStr), topupTotalDiff)) :+ TopupMutexIntEntry(internalBaseAssetStr, 0)) :+ IntE(keyTopupLastTimestamp(internalBaseAssetStr), lastBlock.timestamp)) :+ RemainingLimitsStringEntry(keyLimitsRemaining(internalBaseAssetStr), submitLimitsBaseMax, submitLimitsShareMax)) :+ Burn(shareAssetId, topupTotalDiff[IdxTotalLockedInShare])) :+ Reissue(shareAssetId, -(topupTotalDiff[IdxTotalLockedOutShare]), true)) ++ (if ((0 > income))
                        then [ScriptTransfer(i.caller, -(income), baseAssetId)]
                        else nil))
                    }
    }



@Callable(i)

    let sysStateTuple = privateCurrentSysParamsREST(baseAssetStr)
    let price = sysStateTuple._1.value
    let decimalsMultPrice = sysStateTuple._2.value
    let baseAssetBalance = sysStateTuple._3.value
    let totalLockedBaseAmount = sysStateTuple._4.value
    let baseAssetBalanceWCO = sysStateTuple._5.value

    let currIterTotalInShareAmount = sysStateTuple._8.value
    let totalLockedOutBaseAmount = sysStateTuple._9.value
    let totalLockedOutShareAmount = sysStateTuple._10.value
    let decimalsMultBothAssets = sysStateTuple._11.value
    let priceATH = sysStateTuple._12.value

    }


@Verifier(tx)
func verify () = {
    let targetPublicKey =     match managerPublicKeyOrUnit() {
        case pk: ByteVector => 
            pk
        case _: Unit => 
            tx.senderPublicKey
        case _ => 
            throw("Match error")
    }
    sigVerify(tx.bodyBytes, tx.proofs[0], targetPublicKey)
    }
