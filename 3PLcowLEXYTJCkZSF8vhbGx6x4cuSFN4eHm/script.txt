func _checkpoint () = {
    let prevUnitsPerUSDBalanceIntegral = _loadUnitsPerUSDBalanceIntegral()
    let prevCheckpointAt = max([_loadStartBlock(), _loadCheckpointAt()])
    let prevUnitsBalanceDistributed = _loadUnitsBalanceDistributed()
    let prevUnitsPerReferalShareIntegral = _loadUnitsPerReferalShareIntegral()
    let prevUnitsSharesReferal = _loadUnitsSharesReferal()
    let currentBlock = min([toBigInt(height), _loadFinishBlock()])
    if ((prevCheckpointAt >= currentBlock))
        then $Tuple3(nil, prevUnitsPerUSDBalanceIntegral, prevUnitsPerReferalShareIntegral)
        else {
            let unitsBalanceToDistribute = fraction((_loadTotalUnitsForLP() - prevUnitsBalanceDistributed), (currentBlock - prevCheckpointAt), (_loadFinishBlock() - prevCheckpointAt))
            let unitsPerUSDBalanceIntegral = (((PERCENT_FACTOR * unitsBalanceToDistribute) / _loadTotalUSDBalance()) + prevUnitsPerUSDBalanceIntegral)
            let $t01745618009 = if ((_loadTotalSharesReferal() > ZERO_BIGINT))
                then {
                    let unitsSharesReferal = fraction((_loadTotalUnitsForReferal() - prevUnitsSharesReferal), (currentBlock - prevCheckpointAt), (_loadFinishBlock() - prevCheckpointAt))
                    let unitsPerReferalShareIntegral = (((PERCENT_FACTOR * unitsSharesReferal) / _loadTotalSharesReferal()) + prevUnitsPerReferalShareIntegral)
                    $Tuple2(unitsSharesReferal, unitsPerReferalShareIntegral)
                    }
                else $Tuple2(ZERO_BIGINT, ZERO_BIGINT)
            let unitsSharesReferal = $t01745618009._1
            let unitsPerReferalShareIntegral = $t01745618009._2
            $Tuple3(((((_saveCheckpointAt(currentBlock) ++ _saveUnitsPerUSDBalanceIntegral(unitsPerUSDBalanceIntegral)) ++ _saveUnitsBalanceDistributed((unitsBalanceToDistribute + prevUnitsBalanceDistributed))) ++ _saveUnitsPerReferalShareIntegral(unitsPerReferalShareIntegral)) ++ _saveUnitsSharesReferal((unitsSharesReferal + prevUnitsSharesReferal))), unitsPerUSDBalanceIntegral, unitsPerReferalShareIntegral)
            }
    }


func _userCheckpoint (user_,userUSDBalance_,userSharesReferal_) = {
    let $t01859518707 = _checkpoint()
    let checkpointActions = $t01859518707._1
    let unitsPerUSDBalanceIntegral = $t01859518707._2
    let unitsPerReferalShareIntegral = $t01859518707._3
    let prevUserUnitsPerUSDBalanceIntegral = _loadUserUnitsPerUSDBalanceIntegral(user_)
    let prevUserCheckpointAt = max([_loadStartBlock(), _loadUserCheckpointAt(user_)])
    let prevUserUnitsBalanceDistributed = _loadUserUnitsBalanceDistributed(user_)
    let prevUserUnitsPerReferalShareIntegral = _loadUserUnitsPerReferalShareIntegral(user_)
    let prevUserUnitsSharesReferal = _loadUserUnitsSharesReferal(user_)
    let currentBlock = min([toBigInt(height), _loadFinishBlock()])
    if ((prevUserCheckpointAt >= currentBlock))
        then $Tuple3(checkpointActions, prevUserUnitsBalanceDistributed, prevUserUnitsSharesReferal)
        else {
            let userUnitsBalanceDistributed = (((userUSDBalance_ * (unitsPerUSDBalanceIntegral - prevUserUnitsPerUSDBalanceIntegral)) / PERCENT_FACTOR) + prevUserUnitsBalanceDistributed)
            let userUnitsSharesReferal = (((userSharesReferal_ * (unitsPerReferalShareIntegral - prevUserUnitsPerReferalShareIntegral)) / PERCENT_FACTOR) + prevUserUnitsSharesReferal)
            $Tuple3((((((checkpointActions ++ _saveUserCheckpointAt(user_, currentBlock)) ++ _saveUserUnitsPerUSDBalanceIntegral(user_, unitsPerUSDBalanceIntegral)) ++ _saveUserUnitsBalanceDistributed(user_, userUnitsBalanceDistributed)) ++ _saveUserUnitsPerReferalShareIntegral(user_, unitsPerReferalShareIntegral)) ++ _saveUserUnitsSharesReferal(user_, userUnitsSharesReferal)), userUnitsBalanceDistributed, userUnitsSharesReferal)
            }
    }


@Callable(i)
func init (executor_,oracle_,pauser_,sWavesContract_) = {
    let err = if (if (if (if (if (if (_onlyThisContract(i.caller))
        then _whenNotInitialized()
        else false)
        then _whenMultisigSet()
        else false)
        then _validateAddress(executor_, "init: invalid executor")
        else false)
        then _validateAddress(oracle_, "init: invalid oracle")
        else false)
        then _validateAddress(pauser_, "init: invalid pauser")
        else false)
        then _validateAddress(sWavesContract_, "init: invalid sWavesContract")
        else false
    if ((err == err))
        then $Tuple2(((((_saveInit(true) ++ _saveExecutor(addressFromStringValue(executor_))) ++ _saveOracle(addressFromStringValue(oracle_))) ++ _savePauser(addressFromStringValue(pauser_))) ++ _saveSWavesContract(addressFromStringValue(sWavesContract_))), unit)
        else throw("Strict value is not equal to itself.")
    }



@Callable(i)
func depositWaves (referrer_) = {
    let err = if (if (if (if (_whenInitialized())
        then _whenNotPaused()
        else false)
        then _validatePaymentsSize(i.payments, 1, "depositWaves: no payment")
        else false)
        then _validateWavesPayment(i.payments[0], "depositWaves: invalid asset")
        else false)
        then _validateInt((i.payments[0].amount / 100), 1, MAX_INT, "depositWaves: invalid amount")
        else false
    if ((err == err))
        then {
            let amount = (i.payments[0].amount / 100)
            let sender = toString(i.caller)
            let token = WAVES
            let rate = _getTokenRate(token)
            let $t02661926699 = _loadUserPosition(token, sender)
            let userTokenBalance = $t02661926699._1
            let userTokenBalanceInUSD = $t02661926699._2
            let userUSDBalance = _loadUserUSDBalance(sender)
            let userSharesReferal = _loadUserSharesReferal(sender)
            let $t02681626942 = _userCheckpoint(sender, userUSDBalance, userSharesReferal)
            let checkpointActions = $t02681626942._1
            let userUnitsBalance = $t02681626942._2
            let userUnitsSharesReferal = $t02681626942._3
            let newUserTokenBalance = (userTokenBalance + toBigInt(amount))
            let newUserTokenBalanceInUSD = fraction(newUserTokenBalance, rate, PERCENT_FACTOR, DOWN)
            let newUserUSDBalance = ((userUSDBalance - userTokenBalanceInUSD) + newUserTokenBalanceInUSD)
            let totalBalanceUpdated = ((_loadTotalUSDBalance() - userTokenBalanceInUSD) + newUserTokenBalanceInUSD)
            let usersCount = if ((userUSDBalance == ZERO_BIGINT))
                then (_loadUsersCount() + ONE_BIGINT)
                else _loadUsersCount()
            let refererActions = if (if ((size(referrer_) > 0))
                then _checkRefererExistence(referrer_)
                else false)
                then {
                    let refererAddress = _loadReferer(referrer_)
                    if ((refererAddress == sender))
                        then throw("depositWaves: sender is referrer")
                        else {
                            let sharesToDistribute = fraction(toBigInt(amount), rate, PERCENT_FACTOR, DOWN)
                            let sharesToUser = fraction(sharesToDistribute, toBigInt(20), toBigInt(100), DOWN)
                            let sharesToReferer = (sharesToDistribute - sharesToUser)
                            let newUserSharesReferal = (userSharesReferal + sharesToUser)
                            let refererSharesReferal = _loadUserSharesReferal(refererAddress)
                            let newRefererSharesReferal = (refererSharesReferal + sharesToReferer)
                            let newTotalSharesReferal = (_loadTotalSharesReferal() + sharesToDistribute)
                            let $t02842728598 = _userCheckpoint(refererAddress, _loadUserUSDBalance(refererAddress), refererSharesReferal)
                            let refererCheckpointActions = $t02842728598._1
                            let refererUnitsBalance = $t02842728598._2
                            let refererUnitsSharesReferal = $t02842728598._3
                            (((refererCheckpointActions ++ _saveUserSharesReferal(sender, newUserSharesReferal)) ++ _saveUserSharesReferal(refererAddress, newRefererSharesReferal)) ++ _saveTotalSharesReferal(newTotalSharesReferal))
                            }
                    }
                else nil
            let invocation = invoke(_loadSWavesContract(), FUNC_DEPOSIT, nil, [AttachedPayment(unit, i.payments[0].amount)])
            if ((invocation == invocation))
                then $Tuple2(((((((checkpointActions ++ refererActions) ++ _saveTotalUSDBalance(totalBalanceUpdated)) ++ _saveTokenBalance(WAVES, (_loadTokenBalance(WAVES) + toBigInt(amount)))) ++ _saveUserUSDBalance(sender, newUserUSDBalance)) ++ _saveUsersCount(usersCount)) ++ _saveUserPosition(token, sender, $Tuple2(newUserTokenBalance, newUserTokenBalanceInUSD))), userUnitsBalance)
                else throw("Strict value is not equal to itself.")
            }
        else throw("Strict value is not equal to itself.")
    }

    if ((err == err))

            let $t02988229962 = _loadUserPosition(token, sender)
