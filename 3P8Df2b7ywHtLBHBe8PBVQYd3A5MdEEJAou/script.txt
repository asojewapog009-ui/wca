func checkWithdrawalLimit (address,assetIdStr,amountInternal,sRate) = {
    let withdrawalLimit = valueOrElse(getInteger(this, (("setup_" + assetIdStr) + "_maxWithdrawPer100B")), -1)
    if ((withdrawalLimit == -1))
        then nil
        else {
            let totalSupplied = tryGetInteger(("total_supplied_" + assetIdStr))
            let limitCap = fraction(totalSupplied, withdrawalLimit, 100)
            let lastH = tryGetInteger(((address + "_withdrawLimitHeight_") + assetIdStr))
            let elapsed = (height - lastH)
            let windowExpired = (elapsed > withdrawalLimitBlocks)
            let currentUsed = if (windowExpired)
                then 0
                else tryGetInteger(((address + "_withdrawLimitUsed_") + assetIdStr))
            let blocksToReset = if (windowExpired)
                then withdrawalLimitBlocks
                else (withdrawalLimitBlocks - elapsed)
            let available = (limitCap - currentUsed)
            let availableScaled = fraction(available, sRate, Scale16, CEILING)
            if ((amountInternal > available))
                then throw(((((("withdrawal limit exceeded for " + assetIdStr) + ", available: ") + toString(availableScaled)) + ", blocks to reset: ") + toString(blocksToReset)))
                else ((if (windowExpired)
                    then [IntegerEntry(((address + "_withdrawLimitHeight_") + assetIdStr), height)]
                    else nil) ++ [IntegerEntry(((address + "_withdrawLimitUsed_") + assetIdStr), (currentUsed + amountInternal))])
            }
    }


func checkBorrowLimit (address,assetIdStr,amountInternal,bRate,sRate) = {
    let borrowLimit = valueOrElse(getInteger(this, (("setup_" + assetIdStr) + "_maxBorrowPer100B")), -1)
    if ((borrowLimit == -1))
        then nil
        else {
            let totalSupplied = tryGetInteger(("total_supplied_" + assetIdStr))
            let limitCapSupply = fraction(totalSupplied, borrowLimit, 100)
            let limitCap = fraction(fraction(limitCapSupply, sRate, Scale16), Scale16, bRate)
            let lastH = tryGetInteger(((address + "_borrowLimitHeight_") + assetIdStr))
            let elapsed = (height - lastH)
            let windowExpired = (elapsed > withdrawalLimitBlocks)
            let currentUsed = if (windowExpired)
                then 0
                else tryGetInteger(((address + "_borrowLimitUsed_") + assetIdStr))
            let blocksToReset = if (windowExpired)
                then withdrawalLimitBlocks
                else (withdrawalLimitBlocks - elapsed)
            let available = (limitCap - currentUsed)
            let availableScaled = fraction(available, bRate, Scale16, CEILING)
            if ((amountInternal > available))
                then throw(((((("borrow limit exceeded for " + assetIdStr) + ", available: ") + toString(availableScaled)) + ", blocks to reset: ") + toString(blocksToReset)))
                else ((if (windowExpired)
                    then [IntegerEntry(((address + "_borrowLimitHeight_") + assetIdStr), height)]
                    else nil) ++ [IntegerEntry(((address + "_borrowLimitUsed_") + assetIdStr), (currentUsed + amountInternal))])
            }
    }

                                else unit
                            if ((inv == inv))
                                then ([IntegerEntry(((address + "_supplied_") + assetIdStr), newAddressSupplied), IntegerEntry(("total_supplied_" + assetIdStr), newTotalSupplied)] ++ ratesRecalcResult)
                                else throw("Strict value is not equal to itself.")
                            }

@Callable(i)
func withdraw (assetIdStr,assetAmount) = {
    let $t01906419131 = getActualRate(assetIdStr, "sRate")
    let sRate = $t01906419131._1
    let ratesRecalcResult = $t01906419131._2
    let amount = fraction(assetAmount, Scale16, sRate, CEILING)
    let address = toString(i.caller)
    let assetSupplied = tryGetInteger(("total_supplied_" + assetIdStr))
    let assetBorrowed = tryGetInteger(("total_borrowed_" + assetIdStr))
    let userAssetSupplied = tryGetInteger(((address + "_supplied_") + assetIdStr))
    let userAssetBorrowed = tryGetInteger(((address + "_borrowed_") + assetIdStr))
    let collateralValueInv = invoke(this, "getUserCollateral", [false, address, true, ((assetIdStr + ",supplied,") + toString(-(amount)))], nil)
    if ((collateralValueInv == collateralValueInv))
        then {
            let collateralValue =             match collateralValueInv {
                case x: Int => 
                    x
                case _ => 
                    throw("can't get user collateral value")
            }
            if (!(tryGetBoolean("setup_active")))
                then throw("market is stopped")
                else if ((0 > collateralValue))
                    then throw("you dont have enough collateral for this operation")
                    else if ((amount > (assetSupplied - assetBorrowed)))
                        then throw("this amount is not available on the market")
                        else if ((amount > (userAssetSupplied - userAssetBorrowed)))
                            then throw("this amount is not available for this user")
                            else {
                                let inv = if ((tryGetString(("setup_autostake_" + assetIdStr)) != ""))
                                    then invoke(this, "unstakeToken", [assetIdStr, assetAmount], nil)
                                    else unit
                                if ((inv == inv))
                                    then {
                                        let limitCheckActions = checkWithdrawalLimit(address, assetIdStr, amount, sRate)
                                        if ((limitCheckActions == limitCheckActions))
                                            then (([IntegerEntry(((address + "_supplied_") + assetIdStr), (tryGetInteger(((address + "_supplied_") + assetIdStr)) - amount)), IntegerEntry(("total_supplied_" + assetIdStr), (tryGetInteger(("total_supplied_" + assetIdStr)) - amount)), ScriptTransfer(i.caller, assetAmount, getAssetBytes(assetIdStr))] ++ ratesRecalcResult) ++ limitCheckActions)
                                            else throw("Strict value is not equal to itself.")
                                        }
                                    else throw("Strict value is not equal to itself.")
                                }
            }
        else throw("Strict value is not equal to itself.")
    }
