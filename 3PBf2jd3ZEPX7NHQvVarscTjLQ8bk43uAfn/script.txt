func isAdminCall (i) = if (containsElement([admin1PK, admin2PK, admin3PK], i.callerPublicKey))
    then unit
    else throw("Only admin can call this function")


func isAirdropAllowed (i) = if (if (containsElement([admin1PK, admin2PK, admin3PK], i.callerPublicKey))
    then true
    else containsElement(["3PAKab5QibKS99RSoqx6eZydWqENyfUSBTF"], toBase58String(i.caller.bytes)))
    then unit
    else throw("Only admins can call this function")


func isSelfCall (i) = if ((i.caller == this))
    then unit
    else throw("Only contract itself can call this function")


func isActive () = if (valueOrElse(getBoolean(this, kActive), true))
    then unit
    else throw("DApp is inactive at this moment")


func assetIdToStr (assetId) = match assetId {
    case id: ByteVector => 
        toBase58String(id)
    case waves: Unit => 
        "WAVES"
    case _ => 
        throw("Not Asset id")
}


func claimCalc (user) = {
    let totalTokenStaked = valueOrElse(getInteger(this, kTotalTokenStaked), 0)
    let curInterest = valueOrElse(getInteger(this, kInterest), 0)
    let availableForClaim = valueOrElse(getInteger(this, (user + kUserAvailable)), 0)
    let userInterest = valueOrElse(getInteger(this, (user + kUserInterest)), curInterest)
    let userTokenStaked = valueOrElse(getInteger(this, (user + kUserTokenStaked)), 0)
    let claimAmount = fraction(userTokenStaked, (curInterest - userInterest), SCALE8)
    $Tuple4(curInterest, (claimAmount + availableForClaim), userTokenStaked, totalTokenStaked)
    }


func claimCalcUSDT (user) = {
    let totalTokenStaked = valueOrElse(getInteger(this, kTotalTokenStaked), 0)
    let curInterest = valueOrElse(getInteger(this, kInterestUSDT), 0)
    let availableForClaim = valueOrElse(getInteger(this, (user + kUserAvailableUSDT)), 0)
    let userTokenStaked = valueOrElse(getInteger(this, (user + kUserTokenStaked)), 0)
    let userInterest = valueOrElse(getInteger(this, (user + kUserInterestUSDT)), if ((userTokenStaked > 0))
        then 0
        else curInterest)
    let claimAmount = fraction(userTokenStaked, (curInterest - userInterest), SCALE10)
    $Tuple4(curInterest, (claimAmount + availableForClaim), userTokenStaked, totalTokenStaked)
    }


func claimCalcNft (user) = {
    let totalNftStaked = valueOrElse(getInteger(this, kNFTTotalStaked), 0)
    let curInterest = valueOrElse(getInteger(this, kNFTInterest), 0)
    let availableForClaim = valueOrElse(getInteger(this, (user + kNFTUserAvailableClaim)), 0)
    let userInterest = valueOrElse(getInteger(this, (user + kNFTUserInterest)), curInterest)
    let userNftStaked = valueOrElse(getInteger(this, (user + kuserNFTStaked)), 0)
    let claimAmount = (userNftStaked * (curInterest - userInterest))
    $Tuple4(curInterest, (claimAmount + availableForClaim), userNftStaked, totalNftStaked)
    }


func calcTokenApr () = {
    let lastAirdrop = valueOrElse(getInteger(this, kLastAirdrop), 0)
    let totalTokenStaked = valueOrElse(getInteger(this, kTotalTokenStaked), 0)
    let totalNFTStaked = valueOrElse(getInteger(this, kNFTTotalStaked), 0)
    let tokenAirdrop = if ((totalNFTStaked == 0))
        then lastAirdrop
        else fraction(lastAirdrop, (100 - totalNFTStaked), 100)
    (fraction(tokenAirdrop, PERCENT_SCALE, totalTokenStaked) * 52)
    }


@Callable(i)

    let apr = calcTokenApr()

    $Tuple2(nil, makeString(["%d%d%d", toString(apr), toString(totalTokenStaked), toString(totalNftStaked)], SEPARATOR))
    }

@Verifier(tx)
func verify () = {
    let multiSignedByAdmins = {
        let adminPubKey1Signed = if (sigVerify(tx.bodyBytes, tx.proofs[0], admin1PK))
            then 1
            else 0
        let adminPubKey2Signed = if (sigVerify(tx.bodyBytes, tx.proofs[1], admin2PK))
            then 1
            else 0
        let adminPubKey3Signed = if (sigVerify(tx.bodyBytes, tx.proofs[2], admin3PK))
            then 1
            else 0
        (((adminPubKey1Signed + adminPubKey2Signed) + adminPubKey3Signed) >= 2)
        }
    match tx {
        case inv: InvokeScriptTransaction => 
            let isSelf = sigVerify(tx.bodyBytes, tx.proofs[0], tx.senderPublicKey)
            let isRightFee = if ((inv.fee == 900000))
                then (inv.feeAssetId == unit)
                else false
            let isInitCall = (inv.function == "init")
            let isnoPayments = (size(inv.payments) == 0)
            if (if (if (if (isRightFee)
                then isInitCall
                else false)
                then isSelf
                else false)
                then isnoPayments
                else false)
                then true
                else multiSignedByAdmins
        case _ => 
            multiSignedByAdmins
    }
    }
