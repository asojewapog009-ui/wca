@Callable(i)

@Callable(i)
func nextRound () = {
    let step = valueOrElse(getInteger("lastStep"), 0)
    let lastMove = valueOrElse(getInteger("lastMove"), 0)
    if (((lastMove + Interval) > height))
        then throw("can't start new round, yet")
        else if ((step >= Rounds))
            then throw("last round was already reached")
            else {
                let wavesBal0 = wavesBalance(this).available
                if ((wavesBal0 == wavesBal0))
                    then {
                        let inv1 = invoke(daoAddress, "Claim", nil, nil)
                        if ((inv1 == inv1))
                            then {
                                let wavesAdded = (wavesBalance(this).available - wavesBal0)
                                if ((wavesAdded == wavesAdded))
                                    then {
                                        func doMove (accum,num) =                                         {
                                            let inv2 = invoke(this, "nextRoundForToken", [num, wavesAdded], nil)
                                            if ((inv2 == inv2))
                                                then nil
                                                else throw("Strict value is not equal to itself.")
                                            }

                                        let result = {
                                            let $l = [0, 1, 2, 3]
                                            let $s = size($l)
                                            let $acc0 = nil
                                            func $f0_1 ($a,$i) =                                             if (($i >= $s))
                                                then $a
                                                else doMove($a, $l[$i])

                                            func $f0_2 ($a,$i) =                                             if (($i >= $s))
                                                then $a
                                                else throw("List size exceeds 4")

                                            $f0_2($f0_1($f0_1($f0_1($f0_1($acc0, 0), 1), 2), 3), 4)
                                            }
                                        if ((result == result))
                                            then [IntegerEntry("lastStep", (step + 1)), IntegerEntry("lastMove", (lastMove + Interval))]
                                            else throw("Strict value is not equal to itself.")
                                        }
                                    else throw("Strict value is not equal to itself.")
                                }
                            else throw("Strict value is not equal to itself.")
                        }
                    else throw("Strict value is not equal to itself.")
                }
    }



@Callable(i)
func cancelLastOrder (tokenNum) = if ((i.caller != this))
    then throw("self invoke only")
    else {
        let tokenId = tokens[tokenNum]
        let tokenIdStr = toBase58String(tokenId)
        let lastOrder = getString((tokenIdStr + "_lastOrder"))
        let inv1 = if (if ((lastOrder != unit))
            then (getString(limitsAddress, (("order_" + value(lastOrder)) + "_status")) == "active")
            else false)
            then invoke(limitsAddress, "cancelOrder", [lastOrder, ""], nil)
            else unit
        if ((inv1 == inv1))
            then nil
            else throw("Strict value is not equal to itself.")
        }



@Callable(i)
func nextRoundForToken (tokenNum,wavesAddedForRound) = if ((i.caller != this))
    then throw("self invoke only")
    else {
        let tokenId = tokens[tokenNum]
        let tokenIdStr = toBase58String(tokenId)
        let tokenPrices = prices[tokenNum]
        let tokenBuybackShare = buybackShares[tokenNum]
        let wavesBalance0 = wavesBalance(this).available
        if ((wavesBalance0 == wavesBalance0))
            then {
                let inv1 = invoke(this, "cancelLastOrder", [tokenNum], nil)
                if ((inv1 == inv1))
                    then {
                        let wavesBalance1 = wavesBalance(this).available
                        if ((wavesBalance1 == wavesBalance1))
                            then {
                                let step = (valueOrElse(getInteger((tokenIdStr + "_lastStep")), -1) + 1)
                                let newPrice = prices[tokenNum][step]
                                let wavesBalanceToUse = (fraction(wavesAddedForRound, tokenBuybackShare, 10000) + (wavesBalance1 - wavesBalance0))
                                if ((wavesBalanceToUse == wavesBalanceToUse))
                                    then {
                                        let inv2 = invoke(limitsAddress, "createOrder", [tokenIdStr, fraction(wavesBalanceToUse, tokenScales[tokenNum], newPrice)], [AttachedPayment(unit, wavesBalanceToUse)])
                                        if ((inv2 == inv2))
                                            then {
                                                let orderId = toString(getIntegerValue(limitsAddress, "global_ordersAmount"))
                                                if ((orderId == orderId))
                                                    then [ScriptTransfer(daoAddress, assetBalance(this, tokenId), tokenId), StringEntry((tokenIdStr + "_lastOrder"), orderId), IntegerEntry((tokenIdStr + "_lastStep"), step)]
                                                    else throw("Strict value is not equal to itself.")
                                                }
                                            else throw("Strict value is not equal to itself.")
                                        }
                                    else throw("Strict value is not equal to itself.")
                                }
                            else throw("Strict value is not equal to itself.")
                        }
                    else throw("Strict value is not equal to itself.")
                }
            else throw("Strict value is not equal to itself.")
        }



@Callable(i)
func finishParty () = {
    let step = valueOrElse(getInteger("lastStep"), 0)
    let blockTimeStamp = value(blockInfoByHeight(height)).timestamp
    if ((Rounds > step))
        then throw("last round didn't happen")
        else if ((1798761600000 > blockTimeStamp))
            then throw("Will become available in 2027")
            else {
                func proceedWithToken (accum,tokenNum) =                 {
                    let inv1 = invoke(this, "cancelLastOrder", [tokenNum], nil)
                    if ((inv1 == inv1))
                        then {
                            let tokenBal = assetBalance(this, tokens[tokenNum])
                            if ((tokenBal == tokenBal))
                                then (accum ++ [ScriptTransfer(daoAddress, tokenBal, tokens[tokenNum])])
                                else throw("Strict value is not equal to itself.")
                            }
                        else throw("Strict value is not equal to itself.")
                    }

                let result = {
                    let $l = [0, 1, 2, 3]
                    let $s = size($l)
                    let $acc0 = nil
                    func $f0_1 ($a,$i) =                     if (($i >= $s))
                        then $a
                        else proceedWithToken($a, $l[$i])

                    func $f0_2 ($a,$i) =                     if (($i >= $s))
                        then $a
                        else throw("List size exceeds 4")

                    $f0_2($f0_1($f0_1($f0_1($f0_1($acc0, 0), 1), 2), 3), 4)
                    }
                if ((result == result))
                    then (result ++ [ScriptTransfer(daoAddress, wavesBalance(this).available, unit)])
                    else throw("Strict value is not equal to itself.")
                }
    }


@Verifier(tx)
func verify () = {
    let allowed = valueOrElse(getBoolean(daoAddress, ("3PBVAd4SQzGD7ae8aEVRbNvM8RcrnV8muVT_allow_tx_" + toBase58String(tx.id))), false)
    if (allowed)
        then true
        else {
            let enabled = valueOrElse(getBoolean(configAddress, "op_child_governance_apply_tx_allowed"), true)
            if (!(enabled))
                then throw("tx application throw governance not enabled")
                else (height >= getIntegerValue(votingResultsAddress, ((("proposal_allow_broadcast_" + toString(daoAddress)) + "_") + toBase58String(tx.id))))
            }
    }
