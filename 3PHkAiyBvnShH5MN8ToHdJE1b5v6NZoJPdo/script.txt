func isAdminCall (i) = if (containsElement([admin1PK, admin2PK, admin3PK], i.callerPublicKey))
    then unit
    else throw("Only admin group1 can call this function")


func isSelfCall (i) = if ((i.caller == this))
    then unit
    else throw("Only contract itself can call this function")


func isFarmingCall (i) = if ((i.caller == farmingAddress))
    then unit
    else throw("Only farming contract can call this function")


func parsePercentParamToAmount (param) = {
    let percent =     match param {
        case p: Int => 
            p
        case p: String => 
            valueOrErrorMessage(parseInt(p), "Can't parse percent in params")
        case _ => 
            throw("Match error")
    }
    if (if ((0 >= percent))
        then true
        else (percent >= PERCENT_SCALE))
        then throw("percent must be number between 1(0.1) and 999(99.9)")
        else {
            let amount = fraction(INITIAL_TOKEN_AMOUNT, percent, PERCENT_SCALE)
            amount
            }
    }


func parseLockVestingPeriod (lockStr,vestStr) = {
    let lockPeriod = valueOrErrorMessage(parseInt(lockStr), "Can't parse lock period")
    let vestingPeriod = valueOrErrorMessage(parseInt(vestStr), "Can't parse vesting period")
    if (if ((0 >= lockPeriod))
        then true
        else (0 >= vestingPeriod))
        then throw("lockPeriod and vestingPeriod must be greater than 0")
        else if ((lockPeriod > vestingPeriod))
            then throw("vestingPeriod must be greater than lockPeriod")
            else $Tuple2(lockPeriod, vestingPeriod)
    }


func validateAddress (aStr) = toString(valueOrErrorMessage(addressFromString(aStr), "Wrong address in params"))


func parseWDAOTeamParams (params) = {
    let address = validateAddress(params[0])
    let amount = parsePercentParamToAmount(params[1])
    let $t045414619 = parseLockVestingPeriod(params[2], params[3])
    let lockPeriod = $t045414619._1
    let vestingPeriod = $t045414619._2
    $Tuple4(address, amount, lockPeriod, vestingPeriod)
    }


func parseParams (params) = {
    let address = validateAddress(params[0])
    let amount = parsePercentParamToAmount(params[1])
    $Tuple2(address, amount)
    }


func withdrawTeamDao (totalAmount,lockPeriod,vestingPeriod,withdrawed,address,wKey) = {
    let startHeight = getIntegerValue(this, kStartHeight)
    let lockTill = (startHeight + lockPeriod)
    let canWithdraw = (fraction(totalAmount, min([(height - (startHeight + lockPeriod)), vestingPeriod]), vestingPeriod) - withdrawed)
    if ((lockTill >= height))
        then throw("Lock period is not end")
        else if ((0 >= canWithdraw))
            then throw("Nothing to withdraw")
            else [IntegerEntry(wKey, (withdrawed + canWithdraw)), ScriptTransfer(address, canWithdraw, AXLYId)]
    }


func withdrawToOther (amount,totalAmount,withdrawed,address,wKey) = if ((0 >= amount))
    then throw("amount must be positive value")
    else if (((withdrawed + amount) > totalAmount))
        then throw(((("You can't withdraw more than " + toString(totalAmount)) + " Your balance: ") + toString((totalAmount - withdrawed))))
        else [IntegerEntry(wKey, (withdrawed + amount)), ScriptTransfer(address, amount, AXLYId)]

@Verifier(tx)
func verify () = {
    let multiSignedByAdmins = {
        let adminPubKey1Signed = if (sigVerify(tx.bodyBytes, tx.proofs[0], admin1PK))
            then 1
            else 0
        let adminPubKey2Signed = if (sigVerify(tx.bodyBytes, tx.proofs[1], admin2PK))
            then 1
            else 0
        let adminPubKey3Signed = if (sigVerify(tx.bodyBytes, tx.proofs[2], admin3PK))
            then 1
            else 0
        (((adminPubKey1Signed + adminPubKey2Signed) + adminPubKey3Signed) >= 2)
        }
    match tx {
        case inv: InvokeScriptTransaction => 
            let isSelf = sigVerify(tx.bodyBytes, tx.proofs[0], tx.senderPublicKey)
            let isRightFee = if ((inv.fee == 100900000))
                then (inv.feeAssetId == unit)
                else false
            let isInitCall = (inv.function == "init")
            let isnoPayments = (size(inv.payments) == 0)
            if (if (if (if (isRightFee)
                then isInitCall
                else false)
                then isSelf
                else false)
                then isnoPayments
                else false)
                then true
                else multiSignedByAdmins
        case _ => 
            multiSignedByAdmins
    }
    }
