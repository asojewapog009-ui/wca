func _validateAddress (address_,err_) = match addressFromString(address_) {
    case a: Address =>

    case _ =>

}

    case a: Asset => 
        true
    case _ =>

func _validateInt (val_,lowerBoundary_,upperBoundary_,err_) = if (if ((lowerBoundary_ > val_))
    then true
    else (val_ > upperBoundary_))
    then throw(err_)
    else true

func _normalizeDecimals (amount_,sourceDecimals_,targetDecimals_) = if ((sourceDecimals_ >= targetDecimals_))
    then (amount_ / pow(10, 0, (sourceDecimals_ - targetDecimals_), 0, 0, DOWN))
    else (amount_ * pow(10, 0, (targetDecimals_ - sourceDecimals_), 0, 0, DOWN))

@Verifier(tx)

    case multisig: String => 
        valueOrElse(getBoolean(addressFromStringValue(multisig), makeString([KEY_STATUS, toString(this), toBase58String(tx.id)], SEPARATOR)), false)
