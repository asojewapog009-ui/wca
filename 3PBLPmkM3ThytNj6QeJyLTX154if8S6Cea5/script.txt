func writeConstString (key,value) = if (!(isDefined(getString(this, key))))
    then StringEntry(key, value)
    else throw(("already initialized: " + key))


func writeInt (key,value) = if ((0 > value))
    then throw(((("writing negative value " + toString(value)) + " for key ") + key))
    else IntegerEntry(key, value)


func writeBinary (key,value) = BinaryEntry(key, value)


func changeBy (key,value) = writeInt(key, (valueOrElse(getInteger(this, key), 0) + value))


func asInt (value) = match value {
    case int: Int => 
        int
    case _ => 
        throw("wrong type, expected: Int")
}

    then throw("empty asset id")
    else if ((str == "WAVES"))

    let currentBalance = valueOrElse(getInteger(userSkinStore(child, address)), 0)

        then invoke(withdrawalContract, "slashParked", [child, address, remainder], nil)

    if ((sp == sp))
        then {
            let action = changeBy(userSkinStore(child, address), -(skinSlashed))
            $Tuple2(action, skinSlashed)
            }
        else throw("Strict value is not equal to itself.")
    }

            if ((s == s))
                then if (if ((size(i.payments) != 1))

                    else (i.payments[0].assetId != pwrAssetId))

                                let skinMax = valueOrElse(getInteger(skinMaxStore(child)), 0)
                                let skinRequired = valueOrElse(getInteger(skinRequiredStore(child)), 0)
                                if (if (if ((skinMax > 0))
                                    then true
                                    else (skinRequired == 0))
                                    then (newSkinTotal > skinMax)
                                    else false)
                                    then throw(("Total max. skin is reached: " + toString(skinMax)))
                                    else [changeBy(userSkinStore(child, toString(i.caller)), commitment), IntegerEntry(skinTotalStore1, newSkinTotal), changeBy(pwrTotalStore, commitment)]
                                }
                        }
                else throw("Strict value is not equal to itself.")
            }
        else throw("Strict value is not equal to itself.")
    }



@Callable(i)
func commitForMiner (child,address) = {
    let a = allowed("commitForMiner")
    if ((a == a))
        then if (!(contains(mainAddresses, toString(i.caller))))
            then throw("denied")
            else if ((child != wavesDAO))
                then throw("commitForMiner: child must be wavesDAO")
                else {
                    let p = i.payments[0]
                    if (if ((p.assetId != pwrAssetId))
                        then true
                        else (size(i.payments) != 1))
                        then throw("invalid payments: only $PWR token expected")
                        else {
                            let commitment = p.amount
                            let autocommit = valueOrElse(getBoolean(addressFromStringValue(address), "powerdao_autocommit"), autocommitForMiners)
                            let sharing = isNodeSharing(address)
                            if (sharing)
                                then [changeBy(nodePoolStore(address), commitment), changeBy(nodeTotalRewardedStore(address), commitment)]
                                else if (autocommit)
                                    then [changeBy(userSkinStore(child, address), commitment), changeBy(skinTotalStore(child), commitment), changeBy(pwrTotalStore, commitment)]
                                    else {
                                        let existingPending = valueOrElse(getInteger(pendingCommitAmountStore(address)), 0)
[IntegerEntry(pendingCommitAmountStore(address), (existingPending + commitment))]
                                        }
                            }
                    }
        else throw("Strict value is not equal to itself.")
    }

@Callable(i)
func claimLeaseRewards (node) = {
    let a = allowed("claimLeaseRewards")
    if ((a == a))
        then if (!(isNodeSharing(node)))
            then throw("sharing not enabled for this node")
            else {
                let child = wavesDAO

                let pool = valueOrElse(getInteger(nodePoolStore(node)), 0)
                let claimable = asInt(invoke(addressFromStringValue(node), "processClaim", [user, pool], nil))
                if ((claimable == claimable))
                    then {

                        if ((0 >= amount))
                            then throw("nothing to claim")
                            else [changeBy(userSkinStore(child, user), amount), changeBy(skinTotalStore(child), amount), changeBy(pwrTotalStore, amount), changeBy(nodePoolStore(node), -(amount))]
                        }
                    else throw("Strict value is not equal to itself.")

    }



@Callable(i)
func claimLeaseRewardsForProtocol (node) = {
    let a = allowed("claimLeaseRewards")
    if ((a == a))
        then if (!(isNodeSharing(node)))
            then throw("sharing not enabled for this node")
            else {
                let nodeAddr = addressFromStringValue(node)
                let treasuryAddress = valueOrElse(getString(config, "contract_treasury"), "3PQn4BwNW1kcX1gagVHWxSP5RyQ3V9UYPT8")
                let pool = valueOrElse(getInteger(nodePoolStore(node)), 0)
                let wavesDAOPayout = asInt(invoke(nodeAddr, "processClaim", [wavesDAO, pool], nil))
                if ((wavesDAOPayout == wavesDAOPayout))
                    then {
                        let poolAfterWavesDAO = (pool - wavesDAOPayout)
                        let treasuryPayout = asInt(invoke(nodeAddr, "processClaim", [treasuryAddress, poolAfterWavesDAO], nil))
                        if ((treasuryPayout == treasuryPayout))
                            then {
                                let totalPayout = (wavesDAOPayout + treasuryPayout)
                                if ((0 >= totalPayout))
                                    then throw("nothing to claim")
                                    else {
                                        let wavesDAOTransfer = if ((wavesDAOPayout > 0))
                                            then [ScriptTransfer(addressFromStringValue(wavesDAO), wavesDAOPayout, pwrAssetId)]
                                            else nil
                                        let treasuryTransfer = if ((treasuryPayout > 0))
                                            then [ScriptTransfer(addressFromStringValue(treasuryAddress), treasuryPayout, pwrAssetId)]
                                            else nil
                                        ((wavesDAOTransfer ++ treasuryTransfer) ++ [changeBy(nodePoolStore(node), -(totalPayout))])
                                        }
                                }
                            else throw("Strict value is not equal to itself.")
                        }
                    else throw("Strict value is not equal to itself.")
                }
        else throw("Strict value is not equal to itself.")
    }

@Callable(i)
func invest (child) = {
    let a = allowed("invest")
    if ((a == a))
        then {
            let s = assertNotStarted(child)
            if ((s == s))
                then {
                    let token = investmentsToken(child)
                    if ((i.payments[0].assetId != token))
                        then throw("invalid investments asset id")
                        else {
                            let amount = i.payments[0].amount
                            let maximum = valueOrElse(getInteger(investmentsMaxStore(child)), 0)
                            func datas (a) =                             [changeBy(investorStore(child, toString(i.caller)), a), changeBy(investmentsTotalStore(child), a), changeBy(investmentsPeriodTotalStore(child), a)]

                            if ((maximum > 0))
                                then {
                                    let totalStore = investmentsTotalStore(child)
                                    let total = valueOrElse(getInteger(totalStore), 0)
                                    let investmens = min([amount, (maximum - total)])
                                    if ((0 >= investmens))
                                        then throw(("Total max. investments is reached: " + toString(maximum)))
                                        else {
                                            let chargeBack = (amount - investmens)
                                            if ((chargeBack > 0))
                                                then [ScriptTransfer(i.caller, chargeBack, token), datas(investmens)]
                                                else datas(investmens)
                                            }
                                    }
                                else datas(amount)
                            }
                    }
                else throw("Strict value is not equal to itself.")
            }
        else throw("Strict value is not equal to itself.")
    }

@Callable(i)
func start (child) = {
    let paused = allowed("start")
    if ((paused == paused))
        then {
            let s = if (assertNotStarted(child))
                then true
                else assertNotCancelled(child)
            if ((s == s))
                then {
                    let totalSkin = valueOrElse(getInteger(skinTotalStore(child)), 0)
                    let skinRequired = valueOrErrorMessage(getInteger(skinRequiredStore(child)), "child not registered")
                    if ((skinRequired > totalSkin))
                        then throw("not enough skin in the game")
                        else {
                            let totalInvestments = valueOrElse(getInteger(investmentsTotalStore(child)), 0)
                            if ((getIntegerValue(investmentsRequiredStore(child)) > totalInvestments))
                                then throw("not enough investments")
                                else [IntegerEntry(startHeightStore(child), HEIGHT), ScriptTransfer(addressFromStringValue(treasury(child)), totalInvestments, investmentsToken(child))]
                            }
                    }
                else throw("Strict value is not equal to itself.")
            }
        else throw("Strict value is not equal to itself.")
    }
