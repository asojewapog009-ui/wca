func voteINTERNAL (callerAddressString,keyPrefix,minVotes,voteResult) = {
    let voteKey = keyFullAdminVote(keyPrefix, callerAddressString)
    let adminCurrentVote = getAdminVote(keyPrefix, callerAddressString)
    let err = if (!(isInAdminList(callerAddressString)))
        then throwErr((("Address: " + callerAddressString) + " not in Admin list"))
        else if ((adminCurrentVote == 1))
            then throwErr((voteKey + " you already voted"))
            else unit
    if ((err == err))
        then {
            let votes = countVotes(keyPrefix)
            if (((votes + 1) >= minVotes))
                then {
                    let clearVoteEntries = getClearVoteEntries(keyPrefix)
                    (clearVoteEntries ++ voteResult)
                    }
                else [IntegerEntry(voteKey, 1)]
            }
        else throw("Strict value is not equal to itself.")
    }


func getSwapActions (i,stakingNode) = {
    let userAddress = i.originCaller
    let payment = i.payments[0]
    let assetInAmount = payment.amount
    let assetOutAmount = fraction(assetInAmount, scale8, assetOutPrice)
    let checks = [if ((size(i.payments) == 1))
        then true
        else throwErr("invalid payments size"), if ((payment.assetId == assetInId))
        then true
        else throwErr("invalid payment assetId"), if ((assetOutPrice > 0))
        then true
        else throwErr("invalid assetOutPrice"), if ((assetOutAmount > 0))
        then true
        else throwErr("invalid assetOutAmount")]
    if ((checks == checks))
        then {
            let stakeInvoke = invoke(stakingAddress, "leaseByAddress", [stakingNode, toString(userAddress)], [AttachedPayment(assetOutId, assetOutAmount)])
            if ((stakeInvoke == stakeInvoke))
                then $Tuple2([IntegerEntry(keyStatsTotalIn, (valueOrElse(getInteger(this, keyStatsTotalIn), 0) + assetInAmount)), IntegerEntry(keyStatsTotalOut, (valueOrElse(getInteger(this, keyStatsTotalOut), 0) + assetOutAmount)), IntegerEntry(keyStatsTotalInByAddress(userAddress), (valueOrElse(getInteger(this, keyStatsTotalInByAddress(userAddress)), 0) + assetInAmount)), IntegerEntry(keyStatsTotalOutByAddress(userAddress), (valueOrElse(getInteger(this, keyStatsTotalOutByAddress(userAddress)), 0) + assetOutAmount)), StringEntry(keyHistory(userAddress, i.transactionId), formatHistory(assetInAmount, assetOutAmount, true, stakingNode))], assetOutAmount)
                else throw("Strict value is not equal to itself.")
            }
        else throw("Strict value is not equal to itself.")
    }

@Callable(i)

    let callerAddressString = toBase58String(i.caller.bytes)
    let keyPrefix = keyAllowedTxIdVotePrefix(txId)
    let result = [StringEntry(keyAllowedTxId(), txId)]
    let allowedTxIdOption = getString(this, keyAllowedTxId())
    let err = [if ((size(fromBase58String(txId)) == TXID_BYTES_LENGTH))
        then true
        else throwErr((txId + " is not valid txId")), if (if ((allowedTxIdOption == unit))
        then true
        else (value(allowedTxIdOption) != txId))
        then true
        else throwErr((txId + " is already allowed"))]
    if ((err == err))
        then voteINTERNAL(callerAddressString, keyPrefix, QUORUM, result)
        else throw("Strict value is not equal to itself.")
    }


@Verifier(tx)
func verify () = {
    let byAdmins = (tx.id == fromBase58String(valueOrElse(getString(this, keyAllowedTxId()), "")))
    let byOwner = if ((size(getAdminsList()) >= QUORUM))
        then false
        else sigVerify(tx.bodyBytes, tx.proofs[0], tx.senderPublicKey)
    if (byAdmins)
        then true
        else byOwner
    }
