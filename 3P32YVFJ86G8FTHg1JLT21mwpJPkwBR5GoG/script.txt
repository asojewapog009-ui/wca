func rebalanceLeaseActions (wavesOut,nodeAddr) = {
    let _isPositive = if ((0 > wavesOut))
        then throw("negative amount of WAVES to lease")
        else true
    if ((_isPositive == _isPositive))
        then {
            let currentLeaseId = tryGetString("global_lastLeaseId")
            let cancelLeaseAction = if ((currentLeaseId != ""))
                then [LeaseCancel(fromBase58String(currentLeaseId))]
                else nil
            match nodeAddr {
                case node: Address => 
                    let newLeaseAmount = ((wavesBalance(this).regular - wavesOut) - WAVES_RESERVE)
                    if ((newLeaseAmount > 0))
                        then {
                            let leaseAction = Lease(node, newLeaseAmount)
                            let newLeaseId = toBase58String(calculateLeaseId(leaseAction))
                            if ((newLeaseId == currentLeaseId))
                                then nil
                                else (cancelLeaseAction ++ [leaseAction, StringEntry("global_lastLeaseId", newLeaseId), IntegerEntry("global_lastLeaseAmount", newLeaseAmount)])
                            }
                        else (cancelLeaseAction ++ (if ((currentLeaseId != ""))
                            then [DeleteEntry("global_lastLeaseId"), DeleteEntry("global_lastLeaseAmount")]
                            else nil))
                case _ => 
                    (cancelLeaseAction ++ (if ((currentLeaseId != ""))
                        then [DeleteEntry("global_lastLeaseId"), DeleteEntry("global_lastLeaseAmount")]
                        else nil))
            }
            }
        else throw("Strict value is not equal to itself.")
    }
