func fractionCeil (value,numerator,denominator) = {
    let cand = fraction(value, numerator, denominator)
    let D = 3037000499
    let exact = ((((cand % D) * (denominator % D)) % D) == (((value % D) * (numerator % D)) % D))
    if (exact)
        then cand
        else (cand + 1)
    }

func currentReward (assetId) = {
    let h = syncHeight(assetId)
    let storedReward = valueOrElse(getInteger(this, rewardStore(assetId)), 0)

        else {
            let speed = valueOrElse(getInteger(this, speedStore(assetId)), 0)
            let accumulatedReward = ((HEIGHT - h) * speed)
            (storedReward + accumulatedReward)
            }
    }

@Callable(i)
func lockLP (address,durationBlocks) = {
    let addrCheck = addressFromStringValue(address)
    if ((addrCheck == addrCheck))
        then if (!(started))
            then throw("not started")
            else if ((size(i.payments) == 0))
                then $Tuple2(nil, 0)
                else {
                    let assetId = toBase58String(value(i.payments[0].assetId))
                    let checks = if (mainOnly(i))
                        then opAllowed("lock_lp", assetId)
                        else false
                    if ((checks == checks))
                        then {
                            let amt = i.payments[0].amount
                            let minAmount = valueOrErrorMessage(getInteger(configAddress, ("lock_min_amount_" + assetId)), ("no min_amount for" + assetId))
                            let prevAssets = userAmount(address, assetId)
                            if ((minAmount > (prevAssets + amt)))
                                then throw(((("locking less than min " + toString(minAmount)) + " of ") + assetId))
                                else {
                                    let prevWeightFactor = userWeightFactor(address, assetId)
                                    let prevUserWeight = userWeight(address, assetId)
                                    let prevUnlockHeight = userUnlockHeight(address, assetId)
                                    let weightFactor = valueOrErrorMessage(getInteger(configAddress, ("lock_weight_factor_" + toString(durationBlocks))), ("no weightFactor for durationBlocks " + toString(durationBlocks)))
                                    if (if ((prevWeightFactor > weightFactor))
                                        then (prevUnlockHeight > HEIGHT)
                                        else false)
                                        then throw("can't decrease weightFactor if not unlocked yet")
                                        else {
                                            let newUserWeight = fraction((prevAssets + amt), weightFactor, factorsBase)
                                            let userWeightChange = (newUserWeight - prevUserWeight)
                                            let tokensData = [changeBy(userAmountStore(address, assetId), amt), IntegerEntry(userUnlockHeightStore(address, assetId), (HEIGHT + durationBlocks)), IntegerEntry(userWeightFactorStore(address, assetId), weightFactor), IntegerEntry(userWeightStore(address, assetId), newUserWeight), changeBy(totalAssetLockedStore(assetId), amt), changeBy(totalWeightedAssetLockedStore(assetId), userWeightChange)]
                                            (tokensData ++ updateRewardData(address, assetId, userWeightChange))
                                            }
                                    }
                            }
                        else throw("Strict value is not equal to itself.")
                    }
        else throw("Strict value is not equal to itself.")
    }



@Callable(i)
func cleanUpDistributionFor (address,assetId) = {
    let addrCheck = addressFromStringValue(address)
    if ((addrCheck == addrCheck))
        then if (!(started))
            then throw("not started")
            else {
                let checks = opAllowed("stop_distribution_unlocked_lp", assetId)
                if ((checks == checks))
                    then {
                        let unlockHeight = userUnlockHeight(address, assetId)
                        if ((unlockHeight == 0))
                            then nil
                            else {
                                let userAmountChange = -(userAmount(address, assetId))
                                let userWeightChange = -(userWeight(address, assetId))
                                let exceptionEarlyUnlock = contains(valueOrElse(getString(configAddress, "lp_unlock_free_atoken_ids"), ""), assetId)
                                if (if ((HEIGHT >= unlockHeight))
                                    then true
                                    else exceptionEarlyUnlock)
                                    then {
                                        let tokensData = [DeleteEntry(userUnlockHeightStore(address, assetId)), DeleteEntry(userWeightFactorStore(address, assetId)), DeleteEntry(userWeightStore(address, assetId)), changeBy(totalWeightedAssetLockedStore(assetId), userWeightChange)]
                                        (tokensData ++ updateRewardData(address, assetId, userWeightChange))
                                        }
                                    else throw("not allowed yet")
                                }
                        }
                    else throw("Strict value is not equal to itself.")
                }
        else throw("Strict value is not equal to itself.")
    }

@Verifier(tx)
func verify () = {
    let BASE = 1000
    let quorumRatio = valueOrErrorMessage(getInteger(configAddress, "proposal_quorum_ratio"), "proposalQuorumRatio is not defined")
    let passedRatio = valueOrErrorMessage(getInteger(configAddress, "proposal_passed_ratio"), "proposalThresholdRatio is not defined")
    let gViresContract = valueOrErrorMessage(addressFromString(valueOrErrorMessage(getString(configAddress, "dividends_contract"), "no dividends_contract")), "invalid dividends_contract")
    let votingContract = valueOrErrorMessage(addressFromString(valueOrErrorMessage(getString(configAddress, "voting_contract"), "no voting_contract")), "invalid voting_contract")
    let id = toBase58String(tx.id)
    let votesYes = valueOrElse(getInteger(votingContract, ("proposal_yes_" + id)), 0)
    let votesNo = valueOrElse(getInteger(votingContract, ("proposal_no_" + id)), 0)
    let proposalHeight = valueOrErrorMessage(getInteger(votingContract, ("proposal_height_" + id)), "proposal not registered")
    let applyStart = valueOrElse(getInteger(votingContract, ("proposal_applystart_" + id)), 0)
    let applyEnd = valueOrElse(getInteger(votingContract, ("proposal_applyend_" + id)), 0)
    let totalGVires = valueOrElse(getInteger(votingContract, ("proposal_gvires_" + id)), 0)
    let enabled = valueOrElse(getBoolean(configAddress, "op_governance_apply_tx_paused"), false)
    let voteYes = votesYes
    let voteNo = votesNo
    let totalVotes = (voteYes + voteNo)
    let hasQuorum = (((totalVotes * BASE) / totalGVires) >= quorumRatio)
    let hasPassed = (((voteYes * BASE) / totalVotes) >= passedRatio)
    let tooEarly = (applyStart >= HEIGHT)
    let tooLate = (HEIGHT >= applyEnd)
    let timeDebug = ((((((("proposalHeight = " + toString(proposalHeight)) + ", applyStart = ") + toString(applyStart)) + ", applyEnd = ") + toString(applyEnd)) + ", HEIGHT = ") + toString(HEIGHT))
    let byVoting = if (!(enabled))
        then throw("tx application throw governance not enabled")
        else if (tooEarly)
            then throw(("proposal can't be executed as it's too early: " + timeDebug))
            else if (tooLate)
                then throw(("proposal can't be executed as it's too late:" + timeDebug))
                else if (!(hasQuorum))
                    then throw((((((("no quorum: " + "totalVotes: ") + toString(totalVotes)) + ", totalGVires: ") + toString(totalGVires)) + ", quorumRatio: ") + toString(quorumRatio)))
                    else if (!(hasPassed))
                        then throw((((((("no threshold achieved: " + "voteYes: ") + toString(voteYes)) + ", voteNo: ") + toString(voteNo)) + ", passedRatio: ") + toString(passedRatio)))
                        else true
    if (sigVerify(tx.bodyBytes, tx.proofs[0], tx.senderPublicKey))
        then true
        else byVoting
    }
