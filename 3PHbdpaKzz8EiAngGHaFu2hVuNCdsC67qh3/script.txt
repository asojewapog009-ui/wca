@Callable(i)
func proxyLiquidateV2 (market,userAddress,debtAssetIdStr,suppliedAssetIdStr,amount) = if ((indexOf(markets, market) == unit))
    then throw("given pool address is not a pool")
    else if (!(verifyLiquidatorAccess(i.caller)))
        then throw("no permission to perform liquidation")
        else {
            let inv = invoke(addressFromStringValue(market), "liquidateV2", [false, userAddress, suppliedAssetIdStr], [AttachedPayment(getAssetBytes(debtAssetIdStr), amount)])
            if ((inv == inv))
                then {
                    let userBorrowedAmount = valueOrElse(getInteger(addressFromStringValue(market), ((userAddress + "_borrowed_") + debtAssetIdStr)), 0)
                    if ((0 > userBorrowedAmount))
                        then throw("transaction leads to a negative borrowed amount")
                        else nil
                    }
                else throw("Strict value is not equal to itself.")
            }



@Callable(i)
func proxyLiquidateV3 (market,userAddress,debtAssetIdStr,suppliedAssetIdStr,amount,routeStr) = if ((indexOf(markets, market) == unit))
    then throw("given market address is not a market")
    else if (!(verifyLiquidatorAccess(i.caller)))
        then throw("no permission to perform liquidation")
        else {
            let marketAddress = addressFromStringValue(market)
            let sRate = getIntegerValue(marketAddress, (suppliedAssetIdStr + "_sRate"))
            let sAmount = getIntegerValue(marketAddress, ((userAddress + "_supplied_") + suppliedAssetIdStr))
            let suppliedAmount = fraction(sAmount, sRate, Scale16)
            let lastLiqForMarketKeyStr = (("history_market_" + market) + "_lastLiquidation")
            let lastLiquidationKeyStr = (("history_user_" + userAddress) + "_lastLiquidation")
            let lastLiquidation = valueOrElse(getInteger(this, lastLiquidationKeyStr), 0)
            if ((amount > fraction(MaxShareToLiquidate, suppliedAmount, 10000)))
                then throw("should liquidate smaller part per time")
                else if (((lastLiquidation + LiquidationDelay) > height))
                    then throw("cannot liquidate yet")
                    else {
                        let inv = invoke(marketAddress, "liquidate", [false, userAddress, amount, suppliedAssetIdStr, debtAssetIdStr, routeStr], nil)
                        if ((inv == inv))
                            then {
                                let userBorrowedAmount = valueOrElse(getInteger(marketAddress, ((userAddress + "_borrowed_") + debtAssetIdStr)), 0)
                                let userUsdBalance = getUserBalanceINTERNAL(marketAddress, userAddress, true)
                                if ((0 > userBorrowedAmount))
                                    then throw("transaction leads to a negative borrowed amount")
                                    else if (if (!(BadDebtLiqAllowed))
                                        then (0 >= userUsdBalance)
                                        else false)
                                        then throw("transaction leads to a bad debt")
                                        else {
                                            let $t070737266 =                                             match inv {
                                                case r: Int => 
                                                    $Tuple2(r, debtAssetIdStr)
                                                case _ => 
                                                    $Tuple2(fraction(amount, LiquidatorReward, 10000), suppliedAssetIdStr)
                                            }
                                            let liquidatorReward = $t070737266._1
                                            let rewardAssetIdStr = $t070737266._2
                                            let statsKeyStr = ((("reward_" + toString(i.caller)) + "_") + rewardAssetIdStr)
[IntegerEntry(statsKeyStr, (valueOrElse(getInteger(this, statsKeyStr), 0) + liquidatorReward)), IntegerEntry(lastLiquidationKeyStr, height), IntegerEntry(lastLiqForMarketKeyStr, height)]
                                            }
                                }
                            else throw("Strict value is not equal to itself.")
                        }
            }



@Callable(i)
func payoutReward (addressStr,assetIdStr) = {
    let payoutAccess = valueOrElse(getString("setup_payoutAccess"), "3PMcMiMEs6w56NRGacksXtFG5zS7doE9fpL")
    if ((indexOf(payoutAccess, toString(i.caller)) == unit))
        then throw("no access to this method")
        else {
            let statsKeyStr = ((("reward_" + addressStr) + "_") + assetIdStr)
            let rewardAvailable = valueOrElse(getInteger(this, statsKeyStr), 0)
[ScriptTransfer(addressFromStringValue(addressStr), rewardAvailable, getAssetBytes(assetIdStr)), IntegerEntry(statsKeyStr, 0)]
            }
    }
