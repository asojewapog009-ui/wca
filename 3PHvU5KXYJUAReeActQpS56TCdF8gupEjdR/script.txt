func userOffCliff (user) = {
    let entryHeight = userEntryHeight(user)
    let factor = userFactor(user)
    if ((factor == 1))
        then (HEIGHT > (entryHeight + (4 * halfLife)))
        else if ((factor == 2))
            then (HEIGHT > (entryHeight + (5 * halfLife)))
            else if ((factor == 4))
                then (HEIGHT > (entryHeight + (6 * halfLife)))
                else throw(("bad factor:" + toString(factor)))
    }


func userViresData (user) = {
    let deposited = userDepositedTokens(user)
    let gV = userGVires(user)
    let supposedLocked = min([deposited, gV])
    let locked = if (userOffCliff(user))
        then 0
        else supposedLocked
    let unlocked = (deposited - locked)
    let withdrawn = userWithdrawnTokens(user)
    let withdrawable = (unlocked - withdrawn)
    let inProtocol = (locked + withdrawable)
    $Tuple6(deposited, locked, withdrawable, withdrawn, inProtocol, gV)
    }
