@Callable(i)
func discoverTreasure (treasureKey,toAddr) = if (!(checkAdmin(i.caller)))
    then throw("You are not authorized to perform this operation")
    else {
        let owned = tryGetString((treasureKey + "_Redeem"))
        if ((owned != ""))
            then throw("This treasure has been assigned to another player")
            else [StringEntry((treasureKey + "_Redeem"), toAddr)]
        }



@Callable(i)
func redeemTreasure (treasureKey) = {
    let owned = tryGetString((treasureKey + "_Redeem"))
    if ((owned != toString(i.caller)))
        then throw("This treasure has already been claimed")
        else {
            let treasureVal = tryGetInteger(treasureKey)
[ScriptTransfer(i.caller, treasureVal, shinywggid), DeleteEntry((treasureKey + "_Redeem")), DeleteEntry(treasureKey)]
            }
    }
