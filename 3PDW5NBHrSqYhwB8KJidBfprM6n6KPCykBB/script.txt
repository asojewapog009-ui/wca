func internalDeposit (addressStr,amountIn) = {
    let amount = tryGetInteger((addressStr + "_Deposit"))
[IntegerEntry((addressStr + "_Deposit"), (amount + amountIn))]
    }


func internalPay (addressStr,amount) = {
    let deposited = tryGetInteger((addressStr + "_Deposit"))
    if ((0 > (deposited - amount)))
        then throw("Not enough deposit.")
        else if (((deposited - amount) == 0))
            then [DeleteEntry((addressStr + "_Deposit"))]
            else [IntegerEntry((addressStr + "_Deposit"), (deposited - amount))]
    }


func internalReserve (addressStr,amount) = {
    let reservedAmount = tryGetInteger((addressStr + "_Reserved"))
    IntegerEntry((addressStr + "_Reserved"), (reservedAmount + amount))
    }


func internalTransfer (fromAddress,toAddress,amount) = (internalPay(fromAddress, amount) ++ internalDeposit(toAddress, amount))


func insertRedeem (accum,nextValue) = {
    let val = split(nextValue, "_")
    let addr = val[0]
    let amnt = val[1]
    (accum :+ internalReserve(addr, parseIntValue(amnt)))
    }


@Callable(i)
func Deposit () = {
    let addressStr = toString(i.originCaller)
    let pmt = i.payments[0]
    if ((pmt == pmt))
        then {
            let amountIn = value(i.payments[0].amount)
            if ((amountIn == amountIn))
                then {
                    let assetIn = pmt.assetId
                    if ((assetIn == assetIn))
                        then if ((assetIn != shinywggid))
                            then throw("Only $HEG payments allowed")
                            else {
                                let amount = tryGetInteger((addressStr + "_Deposit"))
                                $Tuple2(internalDeposit(addressStr, amountIn), (amount + amountIn))
                                }
                        else throw("Strict value is not equal to itself.")
                    }
                else throw("Strict value is not equal to itself.")
            }
        else throw("Strict value is not equal to itself.")
    }

@Callable(i)
func Redeem () = {
    let addressStr = toString(i.caller)
    let reservedAmount = tryGetInteger((addressStr + "_Reserved"))
    if ((reservedAmount == 0))
        then throw("Nothing to redeem")
        else [DeleteEntry((addressStr + "_Reserved")), ScriptTransfer(i.caller, reservedAmount, shinywggid)]
    }
