func key_index (asset) = ("x_" + asset)


func key_staking (asset) = ("s_" + asset)


let assets_string = getStringValue(this, key_assets)

let assets = split_4C(assets_string, SEPARATOR)

func check_address (a58) = {
    let a = addressFromStringValue(a58)
    toString(a)
    }


func check_asset (asset58) = if ((asset58 == "WAVES"))
    then "WAVES"
    else {
        let asset = valueOrErrorMessage(fromBase58String(asset58), ("wrong asset encoding: " + asset58))
        let info = valueOrErrorMessage(assetInfo(asset), ("wrong asset info: " + asset58))
        if ((info == info))
            then asset58
            else throw("Strict value is not equal to itself.")
        }


func asset_bytes (a) = if ((a == "WAVES"))
    then unit
    else fromBase58String(a)


func asset_string (a) = match a {
    case b: ByteVector => 
        toBase58String(b)
    case _ => 
        "WAVES"
}


func allowed (caller) = contains(owner, toString(caller))


func exist (asset) = contains(assets_string, asset)


func index (asset) = valueOrErrorMessage(getInteger(key_index(asset)), ("no index for " + asset))


func token_data (index) = $Tuple2(parseBigIntValue(balances[index]), parseBigIntValue(weights[index]))


func staking_info (asset) = {
    let info = valueOrElse(getString(this, key_staking(asset)), "")
    if ((info == ""))
        then ["", "", ""]
        else split(info, SEPARATOR)
    }


func available (asset) = if ((asset == "WAVES"))
    then wavesBalance(this).available
    else assetBalance(this, fromBase58String(asset))


func putx (asset) = {
    let info = staking_info(asset)
    let dapp = info[0]
    let amount = available(asset)
    if (if ((dapp == ""))
        then true
        else (0 >= amount))
        then nil
        else invoke(addressFromStringValue(dapp), info[1], nil, [AttachedPayment(asset_bytes(asset), amount)])
    }


func getx (amount,asset) = {
    let info = staking_info(asset)
    let dapp = info[0]
    if (if ((dapp == ""))
        then true
        else (0 >= amount))
        then nil
        else invoke(addressFromStringValue(dapp), info[2], [amount], nil)
    }
