func getOutdatedUr (assetIdStr) = {
    let down = fraction(tryGetInteger(("total_supplied_" + assetIdStr)), tryGetInteger((assetIdStr + "_sRate")), Scale16)
    if ((down == 0))
        then 0
        else fraction(Scale8, fraction(tryGetInteger(("total_borrowed_" + assetIdStr)), tryGetInteger((assetIdStr + "_bRate")), Scale16), down)
    }


func getInterest (assetIdStr) = {
    let ur = getOutdatedUr(assetIdStr)
    let curve = getRateCurve(assetIdStr)
    let rate = (curve._1 + (if ((curve._3 >= ur))
        then fraction(ur, curve._2, curve._3)
        else (curve._2 + fraction((ur - curve._3), curve._4, (100000000 - curve._3)))))
    max([fraction(rate, Scale8, (dayBlocks * 365)), 1])
    }


func tokenRatesRecalc (assetIdStr) = {
    let interest = getInterest(assetIdStr)
    let ur = getOutdatedUr(assetIdStr)
    let lastRecalcHeight = tryGetInteger("lastRateHeight")
    let lastBRate = max([tryGetInteger((assetIdStr + "_bRate")), Scale16])
    let newBRate = (lastBRate + ((height - lastRecalcHeight) * interest))
    let lastSRate = max([tryGetInteger((assetIdStr + "_sRate")), Scale16])
    let newSRate = (lastSRate + ((((height - lastRecalcHeight) * fraction(interest, ur, Scale8)) * (100 - reserveFund)) / 100))
[IntegerEntry((assetIdStr + "_sRate"), newSRate), IntegerEntry((assetIdStr + "_bRate"), newBRate), IntegerEntry("lastRateHeight", height)]
    }

        then throw("1 payment has to be attached")
        else {
            let assetIdStr = getAssetString(i.payments[0].assetId)
            let assetAmount = i.payments[0].amount
            let $t082858352 = getActualRate(assetIdStr, "sRate")
            let sRate = $t082858352._1
            let ratesRecalcResult = $t082858352._2
            let amount = fraction(assetAmount, Scale16, sRate, DOWN)
            let address = toString(i.caller)
            let maxSupply =             match getString(("setup_maxSupply_" + assetIdStr)) {
                case x: String => 
                    parseIntValue(x)
                case _ => 
                    tryGetInteger(("setup_maxSupply_" + assetIdStr))
            }
            let assetPrice = getTokenPrice(assetIdStr)
            let newAddressSupplied = (tryGetInteger(((address + "_supplied_") + assetIdStr)) + amount)
            let newTotalSupplied = (tryGetInteger(("total_supplied_" + assetIdStr)) + amount)
            let rate = getActualRate(assetIdStr, "sRate")._1
            let assetScale = calcAssetScale(assetIdStr)
            let newTotalSuppliedUsd = fraction(fraction(newTotalSupplied, rate, Scale16), assetPrice._1, assetScale)

                then throw("this asset is not supported by the market")
                else if (if ((maxSupply != 0))
                    then (newTotalSuppliedUsd > maxSupply)
                    else false)
                    then throw("max total supply for this token reached in the pool")
                    else if ((tryGetInteger(("setup_paused_" + assetIdStr)) > 0))
                        then throw("this asset can't be supplied or borrowed")
                        else ([IntegerEntry(((address + "_supplied_") + assetIdStr), newAddressSupplied), IntegerEntry(("total_supplied_" + assetIdStr), newTotalSupplied)] ++ ratesRecalcResult)
            }

    let $t098249891 = getActualRate(assetIdStr, "sRate")
    let sRate = $t098249891._1
    let ratesRecalcResult = $t098249891._2
    let amount = fraction(assetAmount, Scale16, sRate, CEILING)
    let address = toString(i.caller)

    let assetBorrowed = tryGetInteger(("total_borrowed_" + assetIdStr))
    let userAssetSupplied = tryGetInteger(((address + "_supplied_") + assetIdStr))
    let userAssetBorrowed = tryGetInteger(((address + "_borrowed_") + assetIdStr))
    let collateralValueInv = invoke(this, "getUserCollateral", [false, address, true, ((assetIdStr + ",supplied,") + toString(-(amount)))], nil)

        then {
            let collateralValue =             match collateralValueInv {
                case x: Int => 
                    x
                case _ => 
                    throw("can't get user collateral value")
            }
            if (!(tryGetBoolean("setup_active")))
                then throw("market is stopped")
                else if ((0 > collateralValue))
                    then throw("you dont have enough collateral for this operation")
                    else if ((amount > (assetSupplied - assetBorrowed)))
                        then throw("this amount is not available on the market")
                        else if ((amount > (userAssetSupplied - userAssetBorrowed)))
                            then throw("this amount is not available for this user")
                            else ([IntegerEntry(((address + "_supplied_") + assetIdStr), (tryGetInteger(((address + "_supplied_") + assetIdStr)) - amount)), IntegerEntry(("total_supplied_" + assetIdStr), (tryGetInteger(("total_supplied_" + assetIdStr)) - amount)), ScriptTransfer(i.caller, assetAmount, getAssetBytes(assetIdStr))] ++ ratesRecalcResult)
            }
        else throw("Strict value is not equal to itself.")
    }
