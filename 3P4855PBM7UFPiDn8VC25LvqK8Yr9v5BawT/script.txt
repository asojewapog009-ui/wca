func getVotingInfoParts (votingInfo) = {
    let votingInfoParts = split(votingInfo, separator)
    let isRewardExistStr = votingInfoParts[1]
    let isRewardExist = if ((isRewardExistStr == "true"))
        then true
        else false
    let rewardAssetId = votingInfoParts[2]
    let rewardAmount = parseIntValue(votingInfoParts[3])
    let votingType = votingInfoParts[4]
    let status = votingInfoParts[5]
    let votingStartHeight = parseIntValue(votingInfoParts[6])
    let votingEndHeight = parseIntValue(votingInfoParts[7])
    let votesQuorum = parseIntValue(votingInfoParts[8])
    let votesFor = parseIntValue(votingInfoParts[9])
    let votesAgainst = parseIntValue(votingInfoParts[10])
    $Tuple10(isRewardExist, rewardAssetId, rewardAmount, votingType, status, votingStartHeight, votingEndHeight, votesQuorum, votesFor, votesAgainst)
    }


func votingExistChecks (assetId,currentIndex) = {
    let votingInfo = valueOrErrorMessage(getString(keyVotingInfo(assetId, currentIndex)), wrapErr("voting info not found"))
    let votingInfoArray = split(votingInfo, separator)
    let status = votingInfoArray[5]
    let votingEndHeight = valueOrErrorMessage(parseIntValue(votingInfoArray[7]), wrapErr("voting start height not found"))
    let suggestIssuer = valueOrErrorMessage(getString(keySuggestIssuer(assetId, currentIndex)), wrapErr("voting issuer not found"))
    let checks = [if ((status == statusInProgress))
        then true
        else throwErr("no voting in progress"), if ((votingEndHeight > height))
        then true
        else throwErr("voting expired")]
    if ((checks == checks))
        then $Tuple3(status, votingEndHeight, suggestIssuer)
        else throw("Strict value is not equal to itself.")
    }


func calculateReward (voter,assetId,index) = {
    let voteKey = keyVote(assetId, index, voter)
    let lastVote = valueOrErrorMessage(getString(voteKey), wrapErr("you have not voted"))
    let lastVoteParts = split(lastVote, separator)
    let gwxAmount = parseIntValue(lastVoteParts[2])
    let votingInfoStr = valueOrErrorMessage(getString(keyVotingInfo(assetId, index)), wrapErr("voting info not found"))
    let votingParts = getVotingInfoParts(votingInfoStr)
    let status = votingParts._5
    let votesFor = votingParts._9
    let votesAgainst = votingParts._10
    let partOfTheTotalVotesX8 = fraction(gwxAmount, MULT8, (votesFor + votesAgainst))
    let totalVotingReward = valueOrElse(getInteger(keyTotalVotingReward(assetId, index)), 0)
    let voterRewardAmount = fraction(partOfTheTotalVotesX8, totalVotingReward, MULT8, FLOOR)
    let canClaim = if ((status == statusAccepted))
        then (voterRewardAmount > 0)
        else false
    $Tuple2(voterRewardAmount, canClaim)
    }

@Callable(i)
func finalize (assetId) = {
    let currentIndexKey = keyCurrentIndex(assetId)
    let currentIndex = valueOrElse(getInteger(currentIndexKey), 0)
    let votingThresholdAdd = valueOrErrorMessage(getInteger(keyVotingThresholdAdd), wrapErr("votingThresholdAdd not set"))
    let votingThresholdRemove = valueOrErrorMessage(getInteger(keyVotingThresholdRemove), wrapErr("votingThresholdRemove not set"))
    let votingInfoStr = valueOrErrorMessage(getString(keyVotingInfo(assetId, currentIndex)), wrapErr("voting info not found"))
    let votingInfoParts = getVotingInfoParts(votingInfoStr)
    let votingType = votingInfoParts._4
    let status = votingInfoParts._5
    let votingEndHeight = votingInfoParts._7
    let votingQuorum = votingInfoParts._8
    let votesFor = votingInfoParts._9
    let votesAgainst = votingInfoParts._10
    let checks = [if ((status == statusInProgress))
        then true
        else throwErr("voting not in progress"), if ((height >= votingEndHeight))
        then true
        else throwErr("voting not finished"), if (isDefined(getString(keyAssetImage(assetId))))
        then true
        else throwErr("asset image not set")]
    if ((checks == checks))
        then {
            let votingAccepted = if (if (((votesFor + votesAgainst) >= votingQuorum))
                then (votesFor > votesAgainst)
                else false)
                then true
                else false
            let newStatus = if (votingAccepted)
                then statusAccepted
                else statusRejected
            let assetImage = getStringValue(keyAssetImage(assetId))

                        else [invoke(assetsStoreContract, "setVerified", [assetId, false], nil), invoke(assetsStoreContract, "onEliminate", [assetId], nil)]

                    }
                else nil
            if ((isVotingAccepted == isVotingAccepted))
                then {
                    let newVotingInfoValue = votingInfoValue(votingInfoParts._1, votingInfoParts._2, votingInfoParts._3, votingInfoParts._4, newStatus, votingInfoParts._6, votingInfoParts._7, votingInfoParts._8, votingInfoParts._9, votingInfoParts._10)

@Verifier(tx)
func verify () = {
    let targetPublicKey =     match managerPublicKeyOrUnit() {
        case pk: ByteVector => 
            pk
        case _: Unit => 
            tx.senderPublicKey
        case _ => 
            throw("Match error")
    }
    sigVerify(tx.bodyBytes, tx.proofs[0], targetPublicKey)
    }
