{-# STDLIB_VERSION 6 #-}
{-# SCRIPT_TYPE ACCOUNT #-}
{-# CONTENT_TYPE DAPP #-}
let keyMAccPKey = "master_pk"

func getCFAddress () = addressFromStringValue(valueOrErrorMessage(getString(this, "CF_ADDRESS"), "CF_ADDRESS not found"))


func tryGetBoolean (key) = match getBoolean(this, key) {
    case b: Boolean => 
        b
    case _ => 
        false
}


let mAccPKey = fromBase58String(valueOrErrorMessage(getString(getCFAddress(), keyMAccPKey), (keyMAccPKey + " not found")))

let mAccAddr = addressFromPublicKey(mAccPKey)

func getCouponsAddress () = addressFromStringValue(getStringValue(mAccAddr, "COUPONS_ADDRESS"))


let keyTotalCompound = "total_compound"

let keyStakedCompound = "staked_compound"

let keyLastVP = "last_virtual_price"

let VPScale = 100000000

func keyStakedCompoundUser (user) = ((toString(user) + "_") + keyStakedCompound)


func calcVirtualPrice (totalStaked,total) = if ((total == 0))
    then 1
    else if ((totalStaked == 0))
        then 1
        else fraction(total, VPScale, totalStaked)


let Scale = 100000000

func keyStakedAmount (address) = (toString(address) + "_farm_staked")


func keyLastCheckInterest (address) = (toString(address) + "_lastCheck_interest")


func keyEggClaimed (address) = (toString(address) + "_claimed")


let keyGlobalLastInterest = "global_lastCheck_interest"

let keyGlobalStaked = "global_staked"

let keyGlobalEggEarned = "global_earnings"

let kLockedInvestments = "locked_investments"

func isCollectiveFarmLocked () = match getBoolean(getCFAddress(), kLockedInvestments) {
    case b: Boolean => 
        b
    case _ => 
        false
}


func getEggId () = fromBase58String(getStringValue(mAccAddr, "EGG_ASSET_ID"))


func setCFAddressAndInitiate (address) = [StringEntry("CF_ADDRESS", address), IntegerEntry(keyGlobalLastInterest, 1)]


func getShareAssetId () = fromBase58String(getStringValue(getCFAddress(), "SHARE_ASSET_ID"))


func tryGetIntegerExternal (address,key) = match getInteger(address, key) {
    case b: Int => 
        b
    case _ => 
        0
}


func tryGetInteger (key) = tryGetIntegerExternal(this, key)


func tryGetString (key) = match getString(this, key) {
    case a: String => 
        a
    case _ => 
        ""
}


func getVoteHeightKey () = "VOTE_HEIGHT_START"


func getDuration () = if ((tryGetIntegerExternal(mAccAddr, "VOTE_DURATION") == 0))
    then 10000
    else tryGetIntegerExternal(mAccAddr, "VOTE_DURATION")


func getVoteByUserKey (user,height) = ((("VOTE_" + user) + "_") + height)


func getTotalVoteByTypeKey (type,height) = ((("VOTE_TOTAL_" + type) + "_") + height)


func getTotalVoteKey (height) = ("VOTE_TOTAL_" + height)


func resultVoteKey (height) = ("LIQUIDATED_" + height)


func quorumVoteKey (height) = ("QUORUM_" + height)


func claimStakingResult (address) = {
    let currentInterest = tryGetInteger(keyGlobalLastInterest)
    let lastCheckInterest = tryGetInteger(keyLastCheckInterest(address))
    let stakedAmount = tryGetInteger(keyStakedAmount(address))
    let reward = if ((lastCheckInterest > 0))
        then fraction((currentInterest - lastCheckInterest), stakedAmount, Scale)
        else 0
    let transfer = if ((reward > 0))
        then [ScriptTransfer(address, reward, getEggId())]
        else nil
    (transfer ++ [IntegerEntry(keyLastCheckInterest(address), currentInterest), IntegerEntry(keyEggClaimed(address), (tryGetInteger(keyEggClaimed(address)) + reward))])
    }


func handleStakingTopUp (amount) = {
    let currentInterest = tryGetInteger(keyGlobalLastInterest)
    let totalStakedAmount = tryGetInteger(keyGlobalStaked)
    let interestDelta = if ((totalStakedAmount > 0))
        then fraction(amount, Scale, totalStakedAmount)
        else 0
[IntegerEntry(keyGlobalEggEarned, (tryGetInteger(keyGlobalEggEarned) + amount)), IntegerEntry(keyGlobalLastInterest, (currentInterest + interestDelta))]
    }


func addVotePower (caller,h,vote,votePower) = if (if ((h == 0))
    then true
    else (vote == ""))
    then nil
    else if ((height > (h + getDuration())))
        then throw((((((((("CAVP: Voting is finished, please finalize the vote;" + " Arguments: ") + caller) + ", ") + toString(h)) + ", ") + vote) + ", ") + toString(votePower)))
        else {
            let voteTotalByType = tryGetInteger(getTotalVoteByTypeKey(vote, toString(h)))
            let totalVote = tryGetInteger(getTotalVoteKey(toString(h)))
[StringEntry(getVoteByUserKey(caller, toString(h)), vote), IntegerEntry(getTotalVoteByTypeKey(vote, toString(h)), (voteTotalByType + votePower)), IntegerEntry(getTotalVoteKey(toString(h)), (totalVote + votePower))]
            }


func adaptVotePowerStake (caller,votePower) = {
    let voteHeight = tryGetInteger(getVoteHeightKey())
    let voteByUserString = tryGetString(getVoteByUserKey(caller, toString(voteHeight)))
    addVotePower(caller, voteHeight, voteByUserString, votePower)
    }


func isLiquidated () = {
    let voteHeight = tryGetInteger(getVoteHeightKey())
    if ((voteHeight == 0))
        then false
        else tryGetBoolean(resultVoteKey(toString(voteHeight)))
    }


@Callable(i)
func claimRefundStaked () = if ((isLiquidated() == false))
    then throw("CCRS: cf not liquidated")
    else {
        let addressStr = toString(i.caller)
        let stakedAmount = tryGetInteger(keyStakedAmount(i.caller))
        let voteHeight = tryGetInteger(getVoteHeightKey())
        let voteByUserString = tryGetString(getVoteByUserKey(addressStr, toString(voteHeight)))
        let multiplier = if ((voteByUserString == ""))
            then 7
            else 10
        let refund = ((stakedAmount / 10) * multiplier)
        let couponsCall = invoke(getCouponsAddress(), "CFRefund", [addressStr, refund], nil)
        if ((couponsCall == couponsCall))
            then [IntegerEntry(keyStakedAmount(i.caller), 0), Burn(getShareAssetId(), stakedAmount)]
            else throw("Strict value is not equal to itself.")
        }



@Callable(i)
func claimRefundUnstaked () = if ((isLiquidated() == false))
    then throw("CCRU: cf not liquidated")
    else {
        let shareTokenId = getShareAssetId()
        if ((size(i.payments) > 1))
            then throw("CSFT: To many payments added")
            else if ((i.payments[0].assetId != shareTokenId))
                then throw("CSFT: Wrong assetId")
                else {
                    let amount = i.payments[0].amount
                    if ((amount == 0))
                        then throw("CSFT: Please attach positive asset amount!")
                        else {
                            let addressStr = toString(i.caller)
                            let multiplier = 7
                            let refund = ((amount / 10) * multiplier)
                            let couponsCall = invoke(getCouponsAddress(), "CFRefund", [addressStr, refund], nil)
                            if ((couponsCall == couponsCall))
                                then [Burn(getShareAssetId(), amount)]
                                else throw("Strict value is not equal to itself.")
                            }
                    }
        }



@Callable(i)
func startVote () = if ((i.caller != mAccAddr))
    then throw("CSV: Only the admin can start a liquidation vote for now!")
    else if ((tryGetInteger(getVoteHeightKey()) != 0))
        then throw("CSV: There is already a vote running!")
        else [IntegerEntry("VOTE_HEIGHT_START", height)]



@Callable(i)
func voteToLiquidate (vote) = {
    let votePower = tryGetInteger(keyStakedAmount(i.caller))
    if ((votePower == 0))
        then throw("CVTL: Please stake some tokens before you can vote!")
        else {
            let voteAsString = toString(vote)
            let voteHeight = tryGetInteger(getVoteHeightKey())
            if ((height > (voteHeight + getDuration())))
                then throw("CVTL: Voting is finished, please finalize the vote!")
                else {
                    let voteByUserString = tryGetString(getVoteByUserKey(toString(i.caller), toString(voteHeight)))
                    if ((voteByUserString != ""))
                        then throw("CVTL: You can not change your vote!")
                        else addVotePower(toString(i.caller), voteHeight, voteAsString, votePower)
                    }
            }
    }



@Callable(i)
func finalizeVote () = {
    let voteHeight = tryGetInteger(getVoteHeightKey())
    if (((voteHeight + getDuration()) > height))
        then throw("CFV: Voting is not finished!")
        else {
            let shareAssetIdTotal = value(assetInfo(getShareAssetId())).quantity
            let totalStakedAmount = tryGetInteger(keyGlobalStaked)
            let totalVote = tryGetInteger(getTotalVoteKey(toString(voteHeight)))
            let quorum = (totalVote / (shareAssetIdTotal / 100))
            let voteTotalByYes = tryGetInteger(getTotalVoteByTypeKey("true", toString(voteHeight)))
            let voteTotalByNo = tryGetInteger(getTotalVoteByTypeKey("false", toString(voteHeight)))
            let liquidated = if ((35 > quorum))
                then true
                else if ((voteTotalByYes > voteTotalByNo))
                    then true
                    else false
            let resetKey = if (liquidated)
                then nil
                else [IntegerEntry(getVoteHeightKey(), 0)]
            ([IntegerEntry(quorumVoteKey(toString(voteHeight)), quorum), BooleanEntry(resultVoteKey(toString(voteHeight)), liquidated)] ++ resetKey)
            }
    }



@Callable(i)
func topUpReward () = if (isLiquidated())
    then throw("CTUR: CF is liquidated!")
    else {
        let eggAssetId = getEggId()
        if ((i.payments[0].assetId != eggAssetId))
            then throw("CTUP: Wrong assetId, payment should be EGG")
            else {
                let resHandleStaking = handleStakingTopUp(i.payments[0].amount)
                $Tuple2(resHandleStaking, true)
                }
        }



@Callable(i)
func withdrawFarmTokens (amount,compound) = if (isLiquidated())
    then throw("CTUR: CF is liquidated!")
    else if ((size(i.payments) > 0))
        then throw("CWFT: Please don't add payments")
        else {
            let shareTokenId = getShareAssetId()
            if (compound)
                then {
                    let staked = tryGetInteger(keyStakedCompound)
                    let total = tryGetInteger(keyTotalCompound)
                    let vp = calcVirtualPrice(staked, total)
                    let keyStakedCompoundU = keyStakedCompoundUser(i.caller)
                    let personalStaked = tryGetInteger(keyStakedCompoundU)
                    let virtualWd = if ((amount == -1))
                        then personalStaked
                        else fraction(amount, VPScale, vp)
                    let amountWd = if ((amount == -1))
                        then fraction(virtualWd, vp, VPScale)
                        else amount
                    if ((virtualWd > personalStaked))
                        then throw("CWFT: You don't have so much funds to withdraw")
                        else [IntegerEntry(keyStakedCompoundU, (personalStaked - virtualWd)), IntegerEntry(keyStakedCompound, (staked - virtualWd)), IntegerEntry(keyTotalCompound, (total - amountWd)), ScriptTransfer(i.caller, amountWd, shareTokenId), IntegerEntry(keyLastVP, vp), IntegerEntry("last_virt_compound_wd", virtualWd), IntegerEntry("last_amount_compound_wd", amountWd)]
                    }
                else {
                    let addressStr = toString(i.caller)
                    let stakedAmount = tryGetInteger(keyStakedAmount(i.caller))
                    let wdAmount = if ((amount == -1))
                        then stakedAmount
                        else amount
                    if ((wdAmount > stakedAmount))
                        then throw("CWFT: you don't have tokens available")
                        else {
                            let votePower = adaptVotePowerStake(toString(i.caller), -(wdAmount))
                            ((claimStakingResult(i.caller) ++ [IntegerEntry(keyStakedAmount(i.caller), (stakedAmount - wdAmount)), IntegerEntry(keyGlobalStaked, (tryGetInteger(keyGlobalStaked) - wdAmount)), ScriptTransfer(i.caller, wdAmount, shareTokenId), IntegerEntry("last_staking_wd", wdAmount)]) ++ votePower)
                            }
                    }
            }



@Callable(i)
func stakeFarmTokens (compound) = if (isLiquidated())
    then throw("CSFT: CF is liquidated!")
    else {
        let shareTokenId = getShareAssetId()
        if ((size(i.payments) > 1))
            then throw("CSFT: Too many payments added")
            else if ((i.payments[0].assetId != shareTokenId))
                then throw("CSFT: Wrong assetId")
                else {
                    let amount = i.payments[0].amount
                    if ((amount == 0))
                        then throw("CSFT: Please attach positive asset amount!")
                        else if (compound)
                            then throw("CSFT: Compound was disabled!")
                            else {
                                let addressStr = toString(i.caller)
                                let totalStakedAmount = tryGetInteger(keyGlobalStaked)
                                if ((i.payments[0].assetId != shareTokenId))
                                    then throw("CSFT: wrong asset attached")
                                    else {
                                        let votePower = adaptVotePowerStake(toString(i.caller), amount)
                                        ((claimStakingResult(i.caller) ++ [IntegerEntry(keyGlobalStaked, (totalStakedAmount + amount)), IntegerEntry(keyStakedAmount(i.caller), (tryGetInteger(keyStakedAmount(i.caller)) + amount))]) ++ votePower)
                                        }
                                }
                    }
        }



@Callable(i)
func initiateDapp (address) = if (isLiquidated())
    then throw("CID: CF is liquidated!")
    else if ((i.caller != this))
        then throw("CID: Can be called only by the dapp-account")
        else setCFAddressAndInitiate(address)



@Callable(i)
func claimReward () = if (isLiquidated())
    then throw("CCR: CF is liquidated!")
    else if ((size(i.payments) > 0))
        then throw("CCR: Please don't add payments")
        else claimStakingResult(i.caller)
