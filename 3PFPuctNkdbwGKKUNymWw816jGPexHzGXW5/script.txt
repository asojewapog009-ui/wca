func _validateAddress (address_,err_) = match addressFromString(address_) {
    case a: Address =>

func _validateInt (val_,lowerBoundary_,upperBoundary_,err_) = if (if ((lowerBoundary_ > val_))
    then true
    else (val_ > upperBoundary_))

    else true

func _validateString (val_,err_) = if (if ((0 >= size(val_)))

func _normalizeDecimals (amount_,sourceDecimals_,targetDecimals_) = if ((sourceDecimals_ >= targetDecimals_))
    then (amount_ / pow(10, 0, (sourceDecimals_ - targetDecimals_), 0, 0, DOWN))
    else (amount_ * pow(10, 0, (targetDecimals_ - sourceDecimals_), 0, 0, DOWN))

        else false)

        else false)
        then _validateWavesAsset(i.payments[0].assetId, "lockTokens: invalid asset")
        else false)

            let afterPercentFee = binding._6

                else false)

                else false
            if ((err1 == err1))
                then {

                        then afterPercentFee
                        else beforePercentFee
                    let fee = (minFee + fraction(amount, percent, PERCENT_FACTOR))

                                    $Tuple2(referrerFeeAmount, [ScriptTransfer(referrer, referrerFeeAmount, unit)])
                                    }
                                else $Tuple2(0, nil)

                            let referrerActions = $t01208212484._2
                            let amountToSend = (amount - fee)

                            if ((err3 == err3))
                                then {

                                    let normalizedAmount = _normalizeDecimals(amountToSend, WAVES_DECIMALS, DECIMALS)
                                    let normalizedGasless = _normalizeDecimals(gaslessReward_, WAVES_DECIMALS, DECIMALS)

                                                else throw("Strict value is not equal to itself.")
                                            }
                                        else throw("Strict value is not equal to itself.")
                                    }
                                else throw("Strict value is not equal to itself.")

@Verifier(tx)
func verify () = match getString(KEY_MULTISIG) {
    case multisig: String => 
        valueOrElse(getBoolean(addressFromStringValue(multisig), makeString([KEY_STATUS, toString(this), toBase58String(tx.id)], SEPARATOR)), false)
    case _ => 
        sigVerify(tx.bodyBytes, tx.proofs[0], tx.senderPublicKey)
}
