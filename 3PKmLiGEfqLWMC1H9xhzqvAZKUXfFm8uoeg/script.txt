func checkAdditionalPayment (payment) = if (isDefined(payment.assetId))
    then throw("BDCAP: Please attach waves")
    else {
        let feeAmount = getIntegerValue(getOracle(), staticKey_extraFee())
        if ((payment.amount != feeAmount))
            then throw((("BDCAP: Please attach exactly " + toString(feeAmount)) + " amount of wavelets"))
            else [ScriptTransfer(getFeeAggregator(), feeAmount, unit)]
        }

func getDucklingPercentage (ducklingId) = {
    let percentageDuckling = tryGetInteger(keyStartPercentage(ducklingId))
    let base = if ((percentageDuckling != 0))
        then percentageDuckling
        else 5
    toBigInt((base * oneSpicePrecsion))
    }


func getCurrentLevelBigInt (ducklingId) = {
    let kDuckLevel = keyDucklingLevel(ducklingId)
    match getString(kDuckLevel) {
        case s: String => 
            parseBigIntValue(s)
        case _ => 
            match getBoolean((("duckling_" + ducklingId) + "_issuedByFeedCall")) {
                case b: Boolean => 
                    if (if ((b == true))
                        then (tryGetInteger(keyStartPercentage(ducklingId)) == 0)
                        else false)
                        then toBigInt(0)
                        else getDucklingPercentage(ducklingId)
                case _ => 
                    getDucklingPercentage(ducklingId)
            }
    }
    }


func calculateNewDucklingLevel (ducklingId,paymentAmount) = {
    let currentTs = lastBlock.timestamp
    let growth = (fraction(toBigInt(paymentAmount), toBigInt(percentGrowthPrecision), toBigInt(DUCKLINGPRICE)) / toBigInt(100))
    let currentLevel = getCurrentLevelBigInt(ducklingId)
    let newLevel = (currentLevel + growth)
    let result = toString(newLevel)
    $Tuple2(result, [("currentLevel=" + toString(currentLevel)), ("newLevel=" + toString(newLevel)), ("growth=" + toString(growth))])
    }

@Callable(i)
func buyDuckling (refererAddress) = {
    let feeValidate = checkAdditionalPayment(i.payments[0])
    if ((feeValidate == feeValidate))
        then {
            let exactPrice = getDucklingPrice()
            if ((50000000 > exactPrice))
                then throw("BBD: Invalid price from oracle!")
                else {
                    let amountPaidByCoupons = {
                        let @ = invoke(getCouponsAddress(), "useCoupons", [exactPrice], nil)
                        if ($isInstanceOf(@, "Int"))
                            then @
                            else throw(($getType(@) + " couldn't be cast to Int"))
                        }
                    if ((amountPaidByCoupons == amountPaidByCoupons))
                        then {
                            let leftToPay = (exactPrice - amountPaidByCoupons)
                            let payment = if ((leftToPay != 0))
                                then {
                                    let firstPayment = value(i.payments[1])
                                    if ((firstPayment.assetId != getEggAssetId()))
                                        then throw(("BBD: You can attach only EGG tokens with the following asset id: " + toBase58String(getEggAssetId())))
                                        else if ((firstPayment.amount != leftToPay))
                                            then throw(("BBD: To buy a perch you currently need the following amount of EGGlets: " + toString(leftToPay)))
                                            else {
                                                let refererRewardForPerch = fraction(leftToPay, 5, 100)
                                                let refCall = asBoolean(invoke(getRefContractAddress(), "refPayment", [refererAddress], [AttachedPayment(getEggAssetId(), refererRewardForPerch)]))
                                                if ((refCall == refCall))
                                                    then {
                                                        let toBurn = if (refCall)
                                                            then (leftToPay - refererRewardForPerch)
                                                            else leftToPay
                                                        let burnCall = invoke(getBurnAddress(), "burnAttachedPayments", nil, [AttachedPayment(getEggAssetId(), toBurn)])
                                                        if ((burnCall == burnCall))
                                                            then leftToPay
                                                            else throw("Strict value is not equal to itself.")
                                                        }
                                                    else throw("Strict value is not equal to itself.")
                                                }
                                    }
                                else 0
                            if ((payment == payment))
                                then {
                                    let ducklingAssetId = invoke(this, "issueFreeDuckling", [toString(i.originCaller), toBase58String(i.transactionId), 0], nil)
                                    if ((ducklingAssetId == ducklingAssetId))
                                        then feeValidate
                                        else throw("Strict value is not equal to itself.")
                                    }
                                else throw("Strict value is not equal to itself.")
                            }
                        else throw("Strict value is not equal to itself.")
                    }
            }
        else throw("Strict value is not equal to itself.")
    }

@Callable(i)
func feedDuckling (ducklingId,backendSignature,maxFeedAmount,userNonce) = {
    let feeValidate = checkAdditionalPayment(i.payments[1])
    if ((feeValidate == feeValidate))
        then {
            let addressString = toString(i.caller)
            let backendProof = getBackendProof(maxFeedAmount, userNonce, addressString)
            let kAddressNonce = keyAddressNonce(addressString)
            let currentNonce = tryGetInteger(kAddressNonce)
            let realDucklingId = if (!(sigVerify_8Kb(toBytes(backendProof), fromBase58String(backendSignature), backendPubKey)))
                then throw("BFD: Invalid proof from backend")
                else if (if ((size(i.payments) != 2))
                    then true
                    else (value(i.payments[0]).assetId != getSpiceAssetId()))
                    then throw("BFD: Bad payment attached (asset[s] or amount)")
                    else if (getBool(keyDucklingGrown(ducklingId)))
                        then throw("BFD: Duckling is already grown")
                        else if ((userNonce != (currentNonce + 1)))
                            then throw(((("BFD: User Nonce should be " + toString(currentNonce)) + " + 1, while received ") + toString(userNonce)))
                            else {
                                let blackListed = tryGetBoolean(keyBlacklisted(ducklingId))
                                if (blackListed)
                                    then throw("BFD: Can not feed blacklisted duckling!")
                                    else if ((ducklingId == ""))
                                        then throw("BFD: Please buy a duckling first!")
                                        else {
                                            let ducklingIdCheck = value(assetInfo(fromBase58String(ducklingId)))
                                            if ((assetBalance(i.caller, ducklingIdCheck.id) != 1))
                                                then throw("BFD: You're not the owner of the duckling")
                                                else if ((ducklingIdCheck.issuer != this))
                                                    then throw("BFD: Cant find duckling with such id")
                                                    else toBase58String(ducklingIdCheck.id)
                                            }
                                }
            let currentPayment = value(i.payments[0]).amount
            let kNewLevel = keyDucklingLevel(realDucklingId)
            let kTotalFeed = keyTotalFeed(realDucklingId)
            let kTotalFeedUser = keyTotalFeedUser(addressString)
            let totalFeed = tryGetInteger(kTotalFeed)
            let totalFeedUser = tryGetInteger(kTotalFeedUser)
            let kFeedTxStats = ((("duckling_" + realDucklingId) + "_stat_") + toString(lastBlock.timestamp))
            if ((currentPayment > maxFeedAmount))
                then throw(("BFD: Cannot feed duckling for such amount, max feed amount is: " + toString(maxFeedAmount)))
                else {
                    let calculateResults = calculateNewDucklingLevel(realDucklingId, currentPayment)
                    let ducklingOwner = tryGetString(keyOwner(realDucklingId))
                    let topup = invoke(getTurtleStakingAddress(), "topUpReward", nil, [AttachedPayment(getSpiceAssetId(), (currentPayment / 2))])
                    if ((topup == topup))
                        then {
                            let topup2 = invoke(getMutantFarmingAddress(), "topUpReward", ["SPICE", 0], [AttachedPayment(getSpiceAssetId(), (currentPayment / 10))])
                            if ((topup2 == topup2))
                                then {
                                    let recordAction = if ((currentPayment >= 1000000000))
                                        then invoke(getCouponsAddress(), "recordAction", ["FEEDDUCKLING10"], nil)
                                        else nil
                                    if ((recordAction == recordAction))
                                        then if (if ((ducklingOwner != toString(i.caller)))
                                            then !(isTestEnv())
                                            else false)
                                            then ([BooleanEntry(keyBlacklisted(realDucklingId), true)] ++ feeValidate)
                                            else ([IntegerEntry(kAddressNonce, (currentNonce + 1)), IntegerEntry(kTotalFeed, (totalFeed + currentPayment)), IntegerEntry(kTotalFeedUser, (totalFeedUser + currentPayment)), IntegerEntry(kFeedTxStats, currentPayment), StringEntry(kNewLevel, calculateResults._1), StringEntry((kFeedTxStats + "_debug"), makeString(calculateResults._2, ";")), Burn(getSpiceAssetId(), ((currentPayment / 2) - (currentPayment / 10)))] ++ feeValidate)
                                        else throw("Strict value is not equal to itself.")
                                    }
                                else throw("Strict value is not equal to itself.")
                            }
                        else throw("Strict value is not equal to itself.")
                    }
            }
        else throw("Strict value is not equal to itself.")
    }



@Callable(i)

    let feeValidate = checkAdditionalPayment(i.payments[1])
    if ((feeValidate == feeValidate))
        then {
            let addressString = toString(i.caller)
            let backendProof = getBackendProofV2(maxFeedAmount, userNonce, addressString, ducklingId, targetLevel)
            let kAddressNonce = keyAddressNonce(addressString)
            let currentNonce = tryGetInteger(kAddressNonce)
            let realDucklingId = if (!(sigVerify_8Kb(toBytes(backendProof), fromBase58String(backendSignature), backendPubKey)))

                    then true

                        else if ((userNonce != (currentNonce + 1)))
                            then throw(((("BFD2: User Nonce should be " + toString(currentNonce)) + " + 1, while received ") + toString(userNonce)))
                            else {
                                let blackListed = tryGetBoolean(keyBlacklisted(ducklingId))
                                if (blackListed)
                                    then throw("BFD2: Can not feed blacklisted duckling!")
                                    else if ((ducklingId == ""))
                                        then throw("BFD2: Please buy a duckling first!")
                                        else if ((0 >= targetLevel))
                                            then throw("BFD2: Invalid target level")
                                            else {
                                                let ducklingIdCheck = value(assetInfo(fromBase58String(ducklingId)))
                                                if ((assetBalance(i.caller, ducklingIdCheck.id) != 1))
                                                    then throw("BFD2: You're not the owner of the duckling")

                                                        then throw("BFD2: Cant find duckling with such id")
                                                        else toBase58String(ducklingIdCheck.id)
                                                }
                                }
            let currentPayment = value(i.payments[0]).amount
            let kNewLevel = keyDucklingLevel(realDucklingId)
            let kTotalFeed = keyTotalFeed(realDucklingId)
            let kTotalFeedUser = keyTotalFeedUser(addressString)
            let totalFeed = tryGetInteger(kTotalFeed)
            let totalFeedUser = tryGetInteger(kTotalFeedUser)
            let kFeedTxStats = ((("duckling_" + realDucklingId) + "_stat_") + toString(lastBlock.timestamp))
            if ((currentPayment > maxFeedAmount))
                then throw(("BFD2: Cannot feed duckling for such amount, max feed amount is: " + toString(maxFeedAmount)))
                else {
                    let calculateResults = calculateNewDucklingLevel(realDucklingId, currentPayment)
                    let normalizedTargetLevel = toBigInt(fraction(targetLevel, oneSpicePrecsion, 100))
                    let newLevel = parseBigIntValue(calculateResults._1)
                    if ((newLevel > normalizedTargetLevel))
                        then throw(("BFD2: Feeding would exceed target level " + toString(targetLevel)))
                        else {
                            let ducklingOwner = tryGetString(keyOwner(realDucklingId))
                            let topup = invoke(getTurtleStakingAddress(), "topUpReward", nil, [AttachedPayment(getSpiceAssetId(), (currentPayment / 2))])
                            if ((topup == topup))
                                then {
                                    let topup2 = invoke(getMutantFarmingAddress(), "topUpReward", ["SPICE", 0], [AttachedPayment(getSpiceAssetId(), (currentPayment / 10))])
                                    if ((topup2 == topup2))
                                        then {
                                            let recordAction = if ((currentPayment >= 1000000000))
                                                then invoke(getCouponsAddress(), "recordAction", ["FEEDDUCKLING10"], nil)
                                                else nil
                                            if ((recordAction == recordAction))
                                                then if (if ((ducklingOwner != toString(i.caller)))
                                                    then !(isTestEnv())
                                                    else false)
                                                    then ([BooleanEntry(keyBlacklisted(realDucklingId), true)] ++ feeValidate)
                                                    else ([IntegerEntry(kAddressNonce, (currentNonce + 1)), IntegerEntry(kTotalFeed, (totalFeed + currentPayment)), IntegerEntry(kTotalFeedUser, (totalFeedUser + currentPayment)), IntegerEntry(kFeedTxStats, currentPayment), StringEntry(kNewLevel, calculateResults._1), StringEntry((kFeedTxStats + "_debug"), makeString(calculateResults._2, ";")), Burn(getSpiceAssetId(), ((currentPayment / 2) - (currentPayment / 10)))] ++ feeValidate)

                                    }
                                else throw("Strict value is not equal to itself.")
                            }
                    }
            }
        else throw("Strict value is not equal to itself.")
    }

@Callable(i)
func turnDucklingIntoDuck () = {
    let feeValidate = checkAdditionalPayment(i.payments[1])
    if ((feeValidate == feeValidate))
        then {
            let address = toString(i.caller)
            let txId = toBase58String(i.transactionId)
            let lastIssuedDucklingTs = tryGetInteger(KGlobalIssuedTimestamp)
            let fiveMinInMs = if (isTestEnv())
                then 0
                else ((1 * 30) * 1000)
            let timeDiff = (lastBlock.timestamp - (lastIssuedDucklingTs + fiveMinInMs))
            if (if ((lastIssuedDucklingTs > 0))
                then (0 > timeDiff)
                else false)
                then throw((("BTD: Can issue ducklings only once per 30s, please wait for " + toString(fraction(timeDiff, 1, 1000))) + " sec."))
                else if ((size(i.payments) != 2))
                    then throw("BTD: Bad payment attached (asset[s] or amount)")
                    else if ((i.payments[0].amount != 1))
                        then throw("BTD: Please attach a nft!")
                        else {
                            let pmt = value(assetInfo(value(value(i.payments[0]).assetId)))
                            let blackListed = tryGetBoolean(keyBlacklisted(toBase58String(pmt.id)))
                            if (blackListed)
                                then throw("BTD: Can not convert blacklisted duckling!")
                                else {
                                    let adultPriceCall = {
                                        let @ = invoke(this, "calculateDucklingPriceTwoDecimals", nil, nil)
                                        if ($isInstanceOf(@, "Int"))
                                            then @
                                            else throw(($getType(@) + " couldn't be cast to Int"))
                                        }
                                    if ((adultPriceCall == adultPriceCall))
                                        then {
                                            let normalizedPrice = fraction(adultPriceCall, oneSpicePrecsion, 100)
                                            if ((toBigInt(normalizedPrice) > getCurrentLevelBigInt(toBase58String(pmt.id))))
                                                then throw("BTD: Duckling is not grown yet...")
                                                else if ((pmt.issuer != this))
                                                    then throw("BTD: Can use only ducklings from this dApp")
                                                    else {
                                                        let call = invoke(getIncubatorAddress(), "startDuckHatching", [""], nil)
                                                        if ((call == call))
                                                            then {
                                                                let kDucklingGrown = keyDucklingGrown(toBase58String(pmt.id))
                                                                ([BooleanEntry(kDucklingGrown, true), IntegerEntry(KGlobalIssuedTimestamp, lastBlock.timestamp), IntegerEntry(keyAmountOfAdultDuckling(), (tryGetInteger(keyAmountOfAdultDuckling()) + 1))] ++ feeValidate)
                                                                }
                                                            else throw("Strict value is not equal to itself.")
                                                        }
                                            }
                                        else throw("Strict value is not equal to itself.")
                                    }
                            }
            }
        else throw("Strict value is not equal to itself.")
    }

        let thirdUser = base58'BpFWP3p3JgYrrP45xfrKzeMcWMEXoinj4FVPPkUiA8D3'
        let firstUserSigned = if (sigVerify(tx.bodyBytes, tx.proofs[0], firstUser))
