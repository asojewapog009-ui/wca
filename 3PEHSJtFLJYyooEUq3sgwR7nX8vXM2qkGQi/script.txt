func asInt (v) = match v {
    case n: Int => 
        n
    case _ => 
        throw("fail to cast into Int")
}


func asTwoIntsTuple (val) = match val {
    case t2: (Int, Int) => 
        t2
    case _ => 
        throw("fail to cast into (Int, Int)")
}


func getVotingPower (userAddrStrOrEmpty) = {
    let props = split(valueOrElse(getString(stakingContract, keyResProportions()), "0_0_0_0_0_0"), "_")
    func adder (acc,item) =     (acc + parseIntValue(item))

    let totalPower = {
        let $l = props
        let $s = size($l)
        let $acc0 = 0
        func $f0_1 ($a,$i) =         if (($i >= $s))
            then $a
            else adder($a, $l[$i])

        func $f0_2 ($a,$i) =         if (($i >= $s))
            then $a
            else throw("List size exceeds 6")

        $f0_2($f0_1($f0_1($f0_1($f0_1($f0_1($f0_1($acc0, 0), 1), 2), 3), 4), 5), 6)
        }
    let usersPower = valueOrElse(getInteger(stakingContract, keyStakedPiecesByOwner(userAddrStrOrEmpty)), 0)
[totalPower, usersPower]
    }


func fixedPoint (val,decimals) = {
    let tenPow = pow(10, 0, decimals, 0, 0, DOWN)
    let lowPart = toString((val % tenPow))
    let zeroes = drop(toString(tenPow), (1 + size(lowPart)))
    (((toString((val / tenPow)) + ".") + zeroes) + lowPart)
    }

func isBlacklistedPromo1 (addr) = contains(BLACKLISTED1, addr)


func claimAcresInternal (addr,acresAmount,lastClaimTime) = {
    let deltaTime = (finalTime() - lastClaimTime)
    let availRes = fraction(deltaTime, (DAILYRESBYPIECE * acresAmount), (DAYMILLIS * MULT8))
    let canGetUsdt = asInt(invoke(economyContract, "sellResourcesWorldREADONLY", [availRes], nil))
    let gotUsdt = max([0, canGetUsdt])
    (gotUsdt * USDT2ACRES_MULTIPLIER)
    }


func claimInternal (addr,acresAmount,lastClaimTime) = {
    let deltaTime = (finalTime() - lastClaimTime)
    if ((0 > deltaTime))
        then throw(((("Saved timestamp is in future, saved = " + toString(lastClaimTime)) + ", final = ") + toString(finalTime())))
        else {
            let availRes = fraction(deltaTime, (DAILYRESBYPIECE * acresAmount), (DAYMILLIS * MULT8))
            let canGetUsdt = asInt(invoke(economyContract, "sellResourcesWorldREADONLY", [availRes], nil))
            let gotUsdt = if ((0 >= canGetUsdt))
                then 0
                else asTwoIntsTuple(invoke(economyContract, "sellResourcesWorld", [addr, availRes], nil))._1
            (gotUsdt * USDT2ACRES_MULTIPLIER)
            }
    }


@Callable(i)
func constructorV1 (restAddr,unstakedPieces) = if ((i.caller != this))
    then throw("Permission denied")
    else if (isDefined(getBinary(acresAssetIdKey)))
        then throw("Already initialized")
        else {
            let nftAcres = (getVotingPower("")[IdxEffTotal] + unstakedPieces)
            let issuedAmount = (MILLION6 - (nftAcres * MULT8))
            let issue = Issue("ACRES", "WavesLands land acres investment token", issuedAmount, 8, false, unit, 0)
            let assetId = calculateAssetId(issue)
[issue, IntegerEntry(acresIssuedAmountKey, issuedAmount), BinaryEntry(acresAssetIdKey, assetId), StringEntry(keyRestAddress(), restAddr)]
            }



@Callable(i)
func unstakeAcresCallback (amount,address) = if ((toString(i.caller) != acres2AddressStr))
    then throw("Permission denied")
    else if ((0 >= amount))
        then throw("Amount should be positive")
        else {
            let timeKey = keyAcresStakedTimeByUser(address)
            let amountKey = keyAcresStakedAmountByUser(address)
            let oldAcresAmount = valueOrElse(getInteger(amountKey), 0)
            if ((amount > oldAcresAmount))
                then throw(((("You have only " + fixedPoint(oldAcresAmount, 8)) + " ACRES staked, tried to unstake ") + fixedPoint(amount, 8)))
                else {
                    let lockedAmount = valueOrElse(getInteger(keyAcresLockedAmountByUser(address)), 0)
                    let lockedETA = valueOrElse(getInteger(keyAcresLockedEtaByUser(address)), 0)
                    let now = lastBlock.timestamp
                    let unstakableAmount = if (isBlacklistedPromo1(address))
                        then ((oldAcresAmount - lockedAmount) - PROMO1_AMOUNT)
                        else (oldAcresAmount - lockedAmount)
                    if (if ((lockedETA > now))
                        then (amount > unstakableAmount)
                        else false)
                        then throw(((("Only " + fixedPoint(unstakableAmount, 8)) + " ACRES can be unstaked now, wait until ") + toString(lockedETA)))
                        else {
                            let oldTotal = valueOrElse(getInteger(acresStakedTotalKey), 0)
                            let wlgResult = invoke(wlgContract, "onStakeUnstakeLandOld", [address], nil)
                            let claimedAcres = claimInternal(address, oldAcresAmount, valueOrElse(getInteger(timeKey), 0))
                            $Tuple2([IntegerEntry(timeKey, lastBlock.timestamp), IntegerEntry(amountKey, (oldAcresAmount - amount)), IntegerEntry(acresStakedTotalKey, (oldTotal - amount)), ScriptTransfer(i.caller, (amount + claimedAcres), acresAssetId)], $Tuple2((amount + claimedAcres), wlgResult))
                            }
                    }
            }



@Callable(i)
func unstakeAllAcresREADONLY (address) = {
    let amount = valueOrElse(getInteger(keyAcresStakedAmountByUser(address)), 0)
    let claimedAcres = claimAcresInternal(address, amount, valueOrElse(getInteger(keyAcresStakedTimeByUser(address)), 0))
    $Tuple2(nil, $Tuple2(amount, claimedAcres))
    }



@Callable(i)
func requestAcresCallback (amount) = {
    let caller = i.caller
    if (if ((toString(caller) != acres2AddressStr))
        then (caller != stakingContract)
        else false)
        then throw("Permission denied")
        else $Tuple2([ScriptTransfer(caller, amount, acresAssetId)], amount)
    }



@Callable(i)
func sendAcresFor (addr,amount) = if ((i.caller != wlgContract))
    then throw("Permission denied")
    else $Tuple2([ScriptTransfer(addressFromStringValue(addr), amount, acresAssetId)], amount)



@Callable(i)
func claimAcresREADONLY (address) = {
    let timeKey = keyAcresStakedTimeByUser(address)
    let amountKey = keyAcresStakedAmountByUser(address)
    let acresAmount = valueOrElse(getInteger(amountKey), 0)
    let lastTime = valueOrElse(getInteger(timeKey), 0)
    let claimedAcres = claimAcresInternal(address, acresAmount, lastTime)
    $Tuple2(nil, [claimedAcres, lastTime])
    }
