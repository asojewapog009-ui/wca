    else if ((0 > inFeeNoLoan))
        then throw("inFeeNoLoan must be greater than 0")
        else if ((0 > inFeeLoan))
            then throw("inFeeLoan must be greater than 0")
            else if ((0 > capFeeNoLoan))
                then throw("capFeeNoLoan must be greater than 0")
                else if ((0 > capFeeWithLoan))
                    then throw("capFeeWithLoan must be greater than 0")
                    else if ((0 > stoplossFeeNoLoan))
                        then throw("stoplossFeeNoLoan must be greater than 0")
                        else if ((0 > stoplossFeeWithLoan))
                            then throw("stoplossFeeWithLoan must be greater than 0")
                            else [IntegerEntry((poolAddr + kAxlyInFeeWithoutLoan), inFeeNoLoan), IntegerEntry((poolAddr + kAxlyInFeeWithLoan), inFeeLoan), IntegerEntry((poolAddr + kAxlyNoLoanCapFee), capFeeNoLoan), IntegerEntry((poolAddr + kAxlyWithLoanCapFee), capFeeWithLoan), IntegerEntry((poolAddr + kAxlyStopLossNoLoanFee), stoplossFeeNoLoan), IntegerEntry((poolAddr + kAxlyStopLossLoanFee), stoplossFeeWithLoan)]))

@Callable(i)
func activatePool (pool) = valueOrElse(isAdminCall(i), if (!(isDefined(getString(this, (kPool + pool)))))
    then throw("Unknown pool")
    else if (valueOrElse(getBoolean(this, (pool + kPoolActive)), true))
        then throw("Pool already active")
        else [BooleanEntry((pool + kPoolActive), true)])



@Callable(i)
func shutdownPool (pool) = valueOrElse(isAdminCall(i), if (!(isDefined(getString(this, (kPool + pool)))))
    then throw("Unknown pool")
    else if (!(valueOrElse(getBoolean(this, (pool + kPoolActive)), true)))
        then throw("Pool already shutdown")
        else [BooleanEntry((pool + kPoolActive), false)])


@Verifier(tx)
func verify () = {
    let multiSignedByAdmins = {
        let adminPubKey1Signed = if (sigVerify(tx.bodyBytes, tx.proofs[0], group1Admin1PK))
            then 1
            else 0
        let adminPubKey2Signed = if (sigVerify(tx.bodyBytes, tx.proofs[1], group1Admin2PK))
            then 1
            else 0
        let adminPubKey3Signed = if (sigVerify(tx.bodyBytes, tx.proofs[2], group1Admin3PK))
            then 1
            else 0
        (((adminPubKey1Signed + adminPubKey2Signed) + adminPubKey3Signed) >= 2)
        }
    multiSignedByAdmins
    }
