func makePayoutDataKey (payoutId) = ("payoutData_" + toString(payoutId))


func makePayoutStatusKey (payoutId) = ("payoutStatus_" + toString(payoutId))


func makePayoutAdditionalRewardsKey (payoutId) = ("payoutAdditionalRewards_" + toString(payoutId))


func getLastPayout () = getIntegerValue(this, LastPayoutKey)


func getPayoutData (payoutId) = {
    let payoutString = getStringValue(makePayoutDataKey(payoutId))
    let payoutDataList = split(payoutString, "_")
    let startHeight = parseIntValue(payoutDataList[0])
    let endHeight = parseIntValue(payoutDataList[1])
    $Tuple2(startHeight, endHeight)
    }


func getPayoutStatus (payoutId) = valueOrElse(getInteger(makePayoutStatusKey(payoutId)), PayoutEmptyStatus)


func getPayoutAdditionalRewards (payoutId) = valueOrElse(getInteger(makePayoutAdditionalRewardsKey(payoutId)), 0)

            else [StringEntry(ConfigAddressKey, configAddress)]

                else {
                    let payoutStatus = getPayoutStatus(payoutId)

                        then throw((("Wrong payout status: " + toString(payoutStatus)) + ", should be 1"))
                        else {

                                then $Tuple2(wavesBalance(this).available, unit)
                                else {
                                    let assetId = fromBase58String(swapFrom)
                                    $Tuple2(assetBalance(this, assetId), assetId)
                                    }
                            let balanceInBeforeSwap = $t068187067._1
                            let swapFromId = $t068187067._2
                            let swapInvoke = invoke(PuzzleAggregatorAddress, "swapWithReferral", [route, minToReceive, "puzzlenode"], [AttachedPayment(swapFromId, amountIn)])
                            if ((swapInvoke == swapInvoke))
                                then {
                                    let balanceInAfterSwap = if ((swapFrom == WavesId))
                                        then wavesBalance(this).available
                                        else assetBalance(this, value(swapFromId))

                                        then throw(("Rollback aggregator refund: " + toString(refund)))
                                        else nil
                                    }
                                else throw("Strict value is not equal to itself.")
                            }
                    }
            }

@Callable(i)
func finalizePayout (payoutId) = if ((i.caller != DaemonAddress))
    then throw("Access denied")
    else {
        let lastPayout = getLastPayout()
        if ((lastPayout != payoutId))
            then throw(((("Wrong payoutId: " + toString(payoutId)) + ", should be: ") + toString(lastPayout)))
            else {
                let payoutStatus = getPayoutStatus(lastPayout)
                if ((payoutStatus != PayoutInitiatedStatus))
                    then throw((("Wrong payout status: " + toString(payoutStatus)) + ", should be: 1"))
                    else [IntegerEntry(makePayoutStatusKey(lastPayout), PayoutCompletedStatus)]
                }
        }



@Callable(i)
func addAdditionalRewards () = if ((size(i.payments) != 1))
    then throw("At least one payment should be attached")
    else if ((i.payments[0].assetId != unit))
        then throw("Only waves can be attached")
        else {
            let lastPayout = getLastPayout()
            let lastPayoutStatus = getPayoutStatus(lastPayout)
            if ((lastPayoutStatus != PayoutCompletedStatus))
                then throw((("Last payout: " + toString(lastPayout)) + " was not completed"))
                else {
                    let nextPayoutId = (lastPayout + 1)
                    let currentAdditionalRewards = getPayoutAdditionalRewards(nextPayoutId)
                    let newAdditionalRewards = (currentAdditionalRewards + i.payments[0].amount)
[IntegerEntry(makePayoutAdditionalRewardsKey(nextPayoutId), newAdditionalRewards)]
                    }
            }



@Callable(i)
func addMassTransferFees () = if ((size(i.payments) != 1))
    then throw("At least one payment should be attached")
    else if ((i.payments[0].assetId != unit))
        then throw("Only waves can be attached")
        else {
            let currentMassTransferFees = valueOrElse(getInteger(MassTransferFeesKey), 0)
            let newMassTransferFees = (currentMassTransferFees + i.payments[0].amount)
[IntegerEntry(MassTransferFeesKey, newMassTransferFees)]
            }



@Callable(i)
func withdrawMassTransferFees () = if ((i.caller != MaintainerAddress))
    then throw("Access denied")
    else if ((size(i.payments) != 0))
        then throw("Payments are prohibited")
        else {
            let currentMassTransferFees = valueOrElse(getInteger(MassTransferFeesKey), 0)
            if ((currentMassTransferFees == 0))
                then throw("Nothing to withdraw")
                else [IntegerEntry(MassTransferFeesKey, 0), ScriptTransfer(MaintainerAddress, currentMassTransferFees, unit)]
            }


@Verifier(tx)
func verify () = match tx {
    case massTx: MassTransferTransaction => 
        let configAddressStr = getString(this, ConfigAddressKey)
        let configAddress = addressFromStringValue(value(configAddressStr))
        let publicKey = if (isDefined(configAddressStr))
            then valueOrElse(getBinary(configAddress, DaemonPublicKeyKey), tx.senderPublicKey)
            else tx.senderPublicKey
        sigVerify_8Kb(massTx.bodyBytes, massTx.proofs[0], publicKey)
    case invokeTx: InvokeScriptTransaction => 
        let configAddressStr = getString(this, ConfigAddressKey)
        let configAddress = addressFromStringValue(value(configAddressStr))
        let unitsChain = getString(configAddress, ("unitsChainName_" + toString({
            let @ = invokeTx.dApp
            if ($isInstanceOf(@, "Address"))
                then @
                else throw(($getType(@) + " couldn't be cast to Address"))
            })))
        let signerPublicKey = valueOrElse(getBinary(configAddress, SignerPublicKeyKey), tx.senderPublicKey)
        let publicKey = if (isDefined(configAddressStr))
            then if (isDefined(unitsChain))
                then tx.senderPublicKey
                else signerPublicKey
            else tx.senderPublicKey
        sigVerify_8Kb(invokeTx.bodyBytes, invokeTx.proofs[0], publicKey)
    case _ => 
        let configAddressStr = getString(this, ConfigAddressKey)
        let configAddress = addressFromStringValue(value(configAddressStr))
        let publicKey = if (isDefined(configAddressStr))
            then valueOrElse(getBinary(configAddress, SignerPublicKeyKey), tx.senderPublicKey)
            else tx.senderPublicKey
        sigVerify_128Kb(tx.bodyBytes, tx.proofs[0], publicKey)
}
