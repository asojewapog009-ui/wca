func deposit (assetId,amount,stakingAssetId,proxy) = {

    if ((currentAdditionalBalance == currentAdditionalBalance))
        then {
            let currentStakingAssetBalance = getStakingAssetBalanceOrZero(stakingAssetId)
            if ((currentStakingAssetBalance == currentStakingAssetBalance))
                then {
                    let asset = parseAssetId(assetId)
                    if ((amount > 0))
                        then {
                            let depositInvoke = invoke(proxy, "deposit", nil, [AttachedPayment(asset, amount)])
                            if ((depositInvoke == depositInvoke))
                                then                                 match depositInvoke {
                                    case receivedStakingAsset: Int => 
                                        let newAdditionalBalance = (currentAdditionalBalance + amount)
                                        let newStakingAssetBalance = (currentStakingAssetBalance + receivedStakingAsset)
[IntegerEntry(keyAdditionalBalance(assetId), newAdditionalBalance), IntegerEntry(keyStakingAssetBalance(stakingAssetId), newStakingAssetBalance)]
                                    case _ => 
                                        nil
                                }
                                else throw("Strict value is not equal to itself.")
                            }
                        else nil
                    }
                else throw("Strict value is not equal to itself.")
            }
        else throw("Strict value is not equal to itself.")
    }


func withdraw (assetId,amount,stakingAssetId,proxy,proxyRateMul,profitAddress) = {
    let currentAdditionalBalance = getAdditionalBalanceOrZero(assetId)
    if ((currentAdditionalBalance == currentAdditionalBalance))
        then {
            let currentStakingAssetBalance = getStakingAssetBalanceOrZero(stakingAssetId)
            if ((currentStakingAssetBalance == currentStakingAssetBalance))
                then {
                    let currentProxyRate = getRate(proxy)
                    if ((currentProxyRate == currentProxyRate))
                        then {
                            let oldRate = fraction(proxyRateMul, currentAdditionalBalance, currentStakingAssetBalance)
                            let stakingAsset = parseAssetId(stakingAssetId)
                            let oldSendStakingAmount = fraction(proxyRateMul, amount, oldRate)
                            let sendStakingAssetAmount = fraction(proxyRateMul, amount, currentProxyRate)
                            let profitAmount = max([0, (oldSendStakingAmount - sendStakingAssetAmount)])
                            if ((sendStakingAssetAmount > 0))
                                then {
                                    let withdrawInvoke = invoke(proxy, "withdraw", nil, [AttachedPayment(stakingAsset, sendStakingAssetAmount)])
                                    if ((withdrawInvoke == withdrawInvoke))
                                        then                                         match withdrawInvoke {
                                            case receivedAssets: Int => 
                                                let newAdditionalBalance = (currentAdditionalBalance - receivedAssets)
                                                let newStakingAssetBalance = ((currentStakingAssetBalance - sendStakingAssetAmount) - profitAmount)
[IntegerEntry(keyAdditionalBalance(assetId), newAdditionalBalance), IntegerEntry(keyStakingAssetBalance(stakingAssetId), newStakingAssetBalance), ScriptTransfer(profitAddress, profitAmount, parseAssetId(stakingAssetId))]
                                            case _ => 
                                                nil
                                        }
                                        else throw("Strict value is not equal to itself.")
                                    }
                                else nil
                            }
                        else throw("Strict value is not equal to itself.")
                    }
                else throw("Strict value is not equal to itself.")
            }
        else throw("Strict value is not equal to itself.")
    }

func rebalanceInternal (targetRatio,assetId,stakingAssetId,minBalance,proxy,proxyRateMul,profitAddress) = {
    let currentStakingAssetBalance = getStakingAssetBalanceOrZero(stakingAssetId)
    if ((currentStakingAssetBalance == currentStakingAssetBalance))
        then {
            let currentAdditionalBalance = if ((currentStakingAssetBalance != 0))
                then getAdditionalBalanceOrZero(assetId)
                else 0
            if ((currentAdditionalBalance == currentAdditionalBalance))
                then {
                    let leasableTotalBalance = max([0, (getAccBalance(assetId) - minBalance)])
                    let targetAdditionalBalance = fraction(targetRatio, leasableTotalBalance, 100)
                    let diff = (currentAdditionalBalance - targetAdditionalBalance)
                    if ((diff == 0))
                        then nil
                        else if ((0 > diff))
                            then {
                                let sendAssetAmount = -(diff)
                                deposit(assetId, sendAssetAmount, stakingAssetId, proxy)
                                }
                            else {
                                let getAssetAmount = diff
                                withdraw(assetId, getAssetAmount, stakingAssetId, proxy, proxyRateMul, profitAddress)
                                }
                    }
                else throw("Strict value is not equal to itself.")
            }
        else throw("Strict value is not equal to itself.")
    }

func estimatePutOperation (txId58,slippageTolerance,inAmAssetAmt,inAmAssetId,inPrAssetAmt,inPrAssetId,userAddress,isEvaluate,emitLp) = {

    let lpAssetId = fromBase58String(cfg[idxPoolLPAssetId])
    let amAssetIdStr = cfg[idxAmtAssetId]
    let prAssetIdStr = cfg[idxPriceAssetId]
    let iAmtAssetId = cfg[idxIAmtAssetId]
    let iPriceAssetId = cfg[idxIPriceAssetId]
    let amtAssetDcm = parseIntValue(cfg[idxAmtAssetDcm])
    let priceAssetDcm = parseIntValue(cfg[idxPriceAssetDcm])
    let poolStatus = cfg[idxPoolStatus]
    let lpEmission = valueOrErrorMessage(assetInfo(lpAssetId), (("Asset " + toBase58String(lpAssetId)) + " doesn't exist")).quantity
    let inAmAssetIdStr = toBase58String(valueOrElse(inAmAssetId, fromBase58String("WAVES")))
    let inPrAssetIdStr = toBase58String(valueOrElse(inPrAssetId, fromBase58String("WAVES")))
    if (if ((amAssetIdStr != inAmAssetIdStr))
        then true
        else (prAssetIdStr != inPrAssetIdStr))
        then throw("Invalid amt or price asset passed.")
        else {
            let amBalance = if (isEvaluate)
                then getAccBalance(amAssetIdStr)
                else (getAccBalance(amAssetIdStr) - inAmAssetAmt)
            let prBalance = if (isEvaluate)
                then getAccBalance(prAssetIdStr)
                else (getAccBalance(prAssetIdStr) - inPrAssetAmt)
            let inAmAssetAmtX18 = toX18(inAmAssetAmt, amtAssetDcm)
            let inPrAssetAmtX18 = toX18(inPrAssetAmt, priceAssetDcm)
            let userPriceX18 = calcPriceBigInt(inPrAssetAmtX18, inAmAssetAmtX18)
            let amBalanceX18 = toX18(amBalance, amtAssetDcm)
            let prBalanceX18 = toX18(prBalance, priceAssetDcm)
            let res = if ((lpEmission == 0))
                then {
                    let curPriceX18 = zeroBigInt
                    let slippageX18 = zeroBigInt
                    let lpAmtX18 = pow((inAmAssetAmtX18 * inPrAssetAmtX18), 0, toBigInt(5), 1, 0, DOWN)
                    $Tuple5(fromX18(lpAmtX18, scale8), fromX18(inAmAssetAmtX18, amtAssetDcm), fromX18(inPrAssetAmtX18, priceAssetDcm), calcPriceBigInt((prBalanceX18 + inPrAssetAmtX18), (amBalanceX18 + inAmAssetAmtX18)), slippageX18)
                    }
                else {
                    let curPriceX18 = calcPriceBigInt(prBalanceX18, amBalanceX18)
                    let slippageX18 = fraction(absBigInt((curPriceX18 - userPriceX18)), scale18, curPriceX18)
                    let slippageToleranceX18 = toX18(slippageTolerance, scale8)
                    if (if ((curPriceX18 != zeroBigInt))
                        then (slippageX18 > slippageToleranceX18)
                        else false)
                        then throw(((("Price slippage " + toString(slippageX18)) + " exceeded the passed limit of ") + toString(slippageToleranceX18)))
                        else {
                            let lpEmissionX18 = toX18(lpEmission, scale8)
                            let prViaAmX18 = fraction(inAmAssetAmtX18, calcPriceBigIntRound(prBalanceX18, amBalanceX18, CEILING), scale18, CEILING)
                            let amViaPrX18 = fraction(inPrAssetAmtX18, scale18, calcPriceBigIntRound(prBalanceX18, amBalanceX18, FLOOR), CEILING)
                            let expectedAmts = if ((prViaAmX18 > inPrAssetAmtX18))
                                then $Tuple2(amViaPrX18, inPrAssetAmtX18)
                                else $Tuple2(inAmAssetAmtX18, prViaAmX18)
                            let expAmtAssetAmtX18 = expectedAmts._1
                            let expPriceAssetAmtX18 = expectedAmts._2
                            let lpAmtX18 = fraction(lpEmissionX18, expPriceAssetAmtX18, prBalanceX18, FLOOR)
                            $Tuple5(fromX18Round(lpAmtX18, scale8, FLOOR), fromX18Round(expAmtAssetAmtX18, amtAssetDcm, CEILING), fromX18Round(expPriceAssetAmtX18, priceAssetDcm, CEILING), curPriceX18, slippageX18)
                            }
                    }
            let calcLpAmt = res._1
            let calcAmAssetPmt = res._2
            let calcPrAssetPmt = res._3
            let curPrice = fromX18(res._4, scale8)
            let slippageCalc = fromX18(res._5, scale8)
            if ((0 >= calcLpAmt))
                then throw("Invalid calculations. LP calculated is less than zero.")
                else {
                    let emitLpAmt = if (!(emitLp))

@Verifier(tx)
func verify () = {
    let targetPublicKey =     match managerPublicKeyOrUnit() {
        case pk: ByteVector => 
            pk
        case _: Unit => 
            tx.senderPublicKey
        case _ => 
            throw("Match error")
    }
    match tx {
        case order: Order => 
            let matcherPub = getMatcherPubOrFail()
            let $t06941869646 = if (skipOrderValidation())
                then $Tuple2(true, "")
                else validateMatcherOrderAllowed(order)
            let orderValid = $t06941869646._1
            let orderValidInfo = $t06941869646._2
            let matcherValid = sigVerify(order.bodyBytes, order.proofs[1], matcherPub)
            if (if (orderValid)
                then matcherValid
                else false)
                then true
                else throwOrderError(orderValid, orderValidInfo, matcherValid)
        case s: SetScriptTransaction => 
            if (sigVerify(tx.bodyBytes, tx.proofs[0], targetPublicKey))
                then true
                else {
                    let newHash = blake2b256(value(s.script))
                    let allowedHash = fromBase64String(value(getString(factoryContract, keyAllowedLpScriptHash())))
                    let currentHash = scriptHash(this)
                    if ((allowedHash == newHash))
                        then (currentHash != newHash)
                        else false
                    }
        case _ => 
            sigVerify(tx.bodyBytes, tx.proofs[0], targetPublicKey)
    }
    }
