func getI (key) = getInteger(this, key)


func getS (key) = getString(this, key)


func getSV (key) = valueOrErrorMessage(getString(this, key), ((("no string value for key " + key) + " at address ") + toString(this)))


func throwIf (condition,error) = if (condition)
    then throw(error)
    else true


func writeInt (key,value) = if ((0 > value))
    then throw(((("writing negative value " + toString(value)) + " for key ") + key))
    else IntegerEntry(key, value)


func ensureGE0 (value) = if ((0 > value))
    then throw("writing negative value")
    else value


func asInt (value) = match value {
    case int: Int => 
        int
    case _ => 
        throw("r:1")
}


func changeBy (key,value) = writeInt(key, (valueOrElse(getI(key), 0) + value))


func writeString (key,value) = StringEntry(key, value)


func fractionCeil (value,numerator,denominator) = {
    let cand = fraction(value, numerator, denominator)
    let D = 3037000499
    let exact = ((((cand % D) * (denominator % D)) % D) == (((value % D) * (numerator % D)) % D))
    if (exact)
        then cand
        else (cand + 1)
    }

    then min([factorsBase, fraction(storedTotalDebt, factorsBase, storedTotalDeposit)])

let apr = {
    let a = getIntegerValue(configAddress, (assetIdStr + "_APoint"))
    let b = getIntegerValue(configAddress, (assetIdStr + "_BPoint"))
    let c = getIntegerValue(configAddress, (assetIdStr + "_CPoint"))
    let d = getIntegerValue(configAddress, (assetIdStr + "_DPoint"))
    let lineAC = (fraction((a - c), utilization, -(b)) + a)
    let lineCD = (fraction((c - d), (utilization - b), (b - factorsBase)) + c)
    if ((utilization == 0))
        then a
        else if ((utilization == b))
            then c
            else if (if ((b > utilization))
                then true
                else (b == factorsBase))
                then lineAC
                else lineCD
    }

    else fraction(fraction(storedTotalDebt, apr, storedTotalDeposit), (factorsBase - reserveFactor), factorsBase)

        }

func paymentAmount (i,assetId) = {
    let p = i.payments[0].amount
    if ((0 >= p))
        then throw(("Payment is less than min allowed amount: " + toString(p)))
        else if ((i.payments[0].assetId != assetId))
            then throw((" bad asset attached: required " + assetIdStr))
            else p
    }


func syncTotals (additionalDeposit,additionalDebt,additionalReserve,keepAtBalance) = {
    let stakingAction = if ((actualBalance == keepAtBalance))
        then unit
        else if (!(stakingEnabled))
            then unit
            else if ((actualBalance > keepAtBalance))
                then invoke(stakingAddress, "put", nil, [AttachedPayment(assetId, (actualBalance - keepAtBalance))])
                else invoke(stakingAddress, "get", [(keepAtBalance - actualBalance)], nil)
    if ((stakingAction == stakingAction))
        then {
            let deposit = ensureGE0((currentTotalDeposit + additionalDeposit))
            let debt = ensureGE0((currentTotalDebt + additionalDebt))
            let reserve = ensureGE0((currentTotalReserve + additionalReserve))
[writeString(reserveGlobalStore, ((((((((toString(HEIGHT) + "|") + toString(deposit)) + "|") + toString(debt)) + "|") + toString(reserve)) + "|") + toString(currentIndex)))]
            }
        else throw("Strict value is not equal to itself.")
    }


func pow10 (n) = if ((n == 6))
    then 1000000
    else if ((n == 8))
        then 100000000
        else throw(("bad decimals: " + toString(n)))


func assetToUsd (amount) = fraction(amount, oraclePrice, pow10(assetDecimals))


func usdToAsset (amount) = fraction(amount, pow10(assetDecimals), oraclePrice)


func aTokenToAsset (aTokenAmount) = if ((aTokenAmount == 0))
    then 0
    else if ((aTokenCirculation > 0))
        then fraction(aTokenAmount, currentTotalDeposit, aTokenCirculation)
        else aTokenAmount


func assetToAToken (assetAmount) = if ((assetAmount == 0))
    then 0
    else if ((aTokenCirculation > 0))
        then fraction(assetAmount, aTokenCirculation, currentTotalDeposit)
        else assetAmount


func assetToATokenCeil (assetAmount) = if ((assetAmount == 0))
    then 0
    else if ((aTokenCirculation > 0))
        then fractionCeil(assetAmount, aTokenCirculation, currentTotalDeposit)
        else assetAmount

func collapseUser (address,amount) = {
    let debt = currentUserDebt(address)
    let deposit = aTokenToAsset(aTokenBalance(address))
    let maxPossible = min([debt, deposit])
    let amt = if ((-1 > amount))
        then throw("invalid collapse amount")
        else if (if ((maxPossible == 0))
            then true
            else (amount == 0))
            then throw("nothing to collapse")
            else if (if ((amount == -1))
                then true
                else (amount > maxPossible))
                then maxPossible
                else amount
    let removedAtokens = assetToATokenCeil(amt)
    $Tuple2((syncTotals(-(amt), -(amt), 0, 0) ++ [changeBy(aTokenBalanceStore(address), -(removedAtokens)), changeBy(aTokenCirculationStore, -(removedAtokens)), writeInt(debtStore(address), (debt - amt)), writeInt(debtIndexStore(address), currentIndex)]), amt)
    }

                    let newDebt = $t01146411767._1
                    let totalDebtUpdate = $t01146411767._2

                    }
                else throw("Strict value is not equal to itself.")
            }
        else throw("Strict value is not equal to itself.")
    }

func withdrawInternal (i,user,amount,toMain) = {
    let maxWithdraw = ((storedTotalDeposit + storedTotalReserve) - storedTotalDebt)
    let checks = if (if (if (mainOnly(i))
        then liquidityCheck(amount, maxWithdraw, ("funds in use: max=" + toString(maxWithdraw)))
        else false)
        then throwIf((-1 > amount), "invalid amount")
        else false)
        then opAllowed("withdraw")
        else false
    if ((checks == checks))
        then {
            let $t01586516056 = if ((amount == -1))
                then {
                    let atokens = aTokenBalance(user)
                    $Tuple2(atokens, aTokenToAsset(atokens))
                    }
                else $Tuple2(assetToATokenCeil(amount), amount)
            let removedAtokens = $t01586516056._1
            let withdrawAmount = $t01586516056._2
            $Tuple2((syncTotals(-(withdrawAmount), 0, 0, withdrawAmount) ++ [ScriptTransfer(if (toMain)
                then mainContract
                else addressFromStringValue(user), withdrawAmount, assetId), changeBy(aTokenBalanceStore(user), -(removedAtokens)), changeBy(aTokenCirculationStore, -(removedAtokens))]), withdrawAmount)
            }
        else throw("Strict value is not equal to itself.")
    }


func replenishInternal (op,user,i) = {
    let checks = if (mainOnly(i))
        then opAllowed(op)
        else false
    if ((checks == checks))

            let aTokenAmount = paymentAmount(i, aTokenId)
            $Tuple2(((syncTotals(0, 0, 0, 0) ++ [changeBy(aTokenBalanceStore(user), aTokenAmount), Burn(aTokenId, aTokenAmount)]) ++ enableColIfNeeded(user)), aTokenToAsset(aTokenAmount))
            }
        else throw("Strict value is not equal to itself.")
    }


func mintInternal (i,address,amountToMint,to,action) = {
    let userATokenBalance = aTokenBalance(address)
    let amount = if ((amountToMint == -1))

    let checks = if (if (if (mainOnly(i))
        then opAllowed(action)
        else false)

        else false)
        then throwIf((amount > userATokenBalance), ("Trying to mint more than available, max: " + toString(userATokenBalance)))
        else false
    if ((checks == checks))
        then $Tuple2((syncTotals(0, 0, 0, 0) ++ [changeBy(aTokenBalanceStore(address), -(amount)), Reissue(aTokenId, amount, true), ScriptTransfer(addressFromStringValue(to), amount, aTokenId)]), aTokenToAsset(amount))
        else throw("Strict value is not equal to itself.")
    }

@Callable(i)
func adviseUser (user) = {
    let currentDebtUsd = assetToUsd(currentTotalDebt)
    let currentDepositUsd = assetToUsd(currentTotalDeposit)
    let asCollateral = valueOrElse(getBoolean(this, useAsCollateralStore(user)), true)
    let effectiveDepositUsd = if (asCollateral)
        then currentDepositUsd
        else 0
    let overlapUsd = min([currentDebtUsd, effectiveDepositUsd])
    let overlapCharge = fractionCeil(overlapUsd, accountHealthOverlap, factorsBase)
    let bp = if ((currentDebtUsd > effectiveDepositUsd))
        then 0
        else fraction((effectiveDepositUsd - currentDebtUsd), collateralFactor, factorsBase)
    let bpu = if ((currentDebtUsd > effectiveDepositUsd))
        then (fraction((currentDebtUsd - effectiveDepositUsd), factorsBase, liquidationThreshold) + overlapCharge)
        else overlapCharge
    let enriched = ((((((((("reserveAddress: " + toString(this)) + ", currentDebtUsd: ") + toString(currentDebtUsd)) + ", currentDepositUsd: ") + toString(currentDepositUsd)) + ", bp: ") + toString(bp)) + ", bpu: ") + toString(bpu))
    $Tuple2(nil, ((enriched + ", ") + getUserState(user)))
    }

func borrowFor (address,amountToBorrow) = {
    let checks = if (if (mainOnly(i))
        then liquidityCheck(amountToBorrow, (storedTotalDeposit - storedTotalDebt), "too much borrow requested")
        else false)
        then opAllowed("borrow")
        else false
    if ((checks == checks))
        then {
            let currentDebt = currentUserDebt(address)
            let newDebt = (currentDebt + amountToBorrow)
            (syncTotals(0, amountToBorrow, 0, amountToBorrow) ++ [writeInt(debtStore(address), newDebt), enableCol(address), writeInt(debtIndexStore(address), currentIndex), ScriptTransfer(addressFromStringValue(address), amountToBorrow, assetId)])
            }
        else throw("Strict value is not equal to itself.")
    }



@Callable(i)
func mintAtokenFor (address,amountToMint) = mintInternal(i, address, amountToMint, address, "mint_atokens")



@Callable(i)
func mintAtokenForTo (from,amountToMint,to) = mintInternal(i, from, amountToMint, to, "lock_mint_atokens")



@Callable(i)
func redeemAtokensFor (user) = {
    let checks = if (mainOnly(i))
        then opAllowed("redeem_atokens")
        else false
    if ((checks == checks))
        then {
            let aTokenAmount = paymentAmount(i, aTokenId)
            let outAmount = aTokenToAsset(aTokenAmount)
            $Tuple2((syncTotals(-(outAmount), 0, 0, outAmount) ++ [ScriptTransfer(addressFromStringValue(user), outAmount, assetId), changeBy(aTokenCirculationStore, -(aTokenAmount)), Burn(aTokenId, aTokenAmount)]), outAmount)
            }
        else throw("Strict value is not equal to itself.")
    }



@Callable(i)
func transferCollateralAndDebt (from,to) = {
    let checks = if (mainOnly(i))
        then opAllowed("assume_full")
        else false
    if ((checks == checks))
        then {
            let asCollateral = valueOrElse(getBoolean(this, useAsCollateralStore(from)), true)
            let aTokens = if (asCollateral)
                then aTokenBalance(from)
                else 0
            let currentDebt = currentUserDebt(from)
            $Tuple2((syncTotals(0, 0, 0, 0) ++ [writeInt(aTokenBalanceStore(from), 0), changeBy(aTokenBalanceStore(to), aTokens), writeInt(debtStore(from), 0), writeInt(debtStore(to), (currentUserDebt(to) + currentDebt)), writeInt(debtIndexStore(from), currentIndex), writeInt(debtIndexStore(to), currentIndex), enableCol(to)]), $Tuple2(aTokens, currentDebt))
            }
        else throw("Strict value is not equal to itself.")
    }



@Callable(i)
func transferATokensFor (from,to,valueUsd) = {
    let checks = if (mainOnly(i))
        then opAllowed("transfer_debt")
        else false
    if ((checks == checks))
        then {
            let assets = usdToAsset(valueUsd)
            let atokens = assetToAToken(assets)
            let aTokensFrom = aTokenBalance(from)
            if ((atokens > aTokensFrom))
                then throw((((((((((("transferAtokensFor error:" + " transfer.valueUsd: ") + toString(valueUsd)) + " transfer.assets: ") + toString(assets)) + " transfer.atokens: ") + toString(atokens)) + " from.atokens: ") + toString(aTokensFrom)) + " at ") + toString(this)))
                else $Tuple2(((syncTotals(0, 0, 0, 0) ++ [changeBy(aTokenBalanceStore(from), -(atokens)), changeBy(aTokenBalanceStore(to), atokens)]) ++ enableColIfNeeded(to)), aTokenToAsset(atokens))
            }
        else throw("Strict value is not equal to itself.")
    }



@Callable(i)
func transferDebtFor (from,to,amount) = {
    let checks = if (mainOnly(i))
        then opAllowed("transfer_debt")
        else false
    if ((checks == checks))
        then $Tuple2((syncTotals(0, 0, 0, 0) ++ [writeInt(debtStore(from), (currentUserDebt(from) - amount)), writeInt(debtStore(to), (currentUserDebt(to) + amount)), writeInt(debtIndexStore(from), currentIndex), writeInt(debtIndexStore(to), currentIndex), enableCol(to)]), amount)
        else throw("Strict value is not equal to itself.")
    }



@Callable(i)
func disableUseAsCollateralFor (address) = {
    let checks = if (mainOnly(i))
        then opAllowed("use_as_col")
        else false
    if ((checks == checks))
        then if ((currentUserDebt(address) > 0))
            then throw("can't disable collateral for asset with open debt")
            else (syncTotals(0, 0, 0, 0) ++ [BooleanEntry(useAsCollateralStore(address), false)])
        else throw("Strict value is not equal to itself.")
    }



@Callable(i)
func enableUseAsCollateral () = {
    let checks = opAllowed("use_as_col")
    if ((checks == checks))
        then (syncTotals(0, 0, 0, 0) ++ [enableCol(toString(i.caller))])
        else throw("Strict value is not equal to itself.")
    }



@Callable(i)
func collapseFor (user) = {
    let checks = if (mainOnly(i))
        then opAllowed("force_collapse")
        else false
    if ((checks == checks))
        then collapseUser(user, -1)
        else throw("Strict value is not equal to itself.")
    }



@Callable(i)
func collapseForAmount (user,amount) = {
    let checks = if (mainOnly(i))
        then opAllowed("collapse")
        else false
    if ((checks == checks))
        then collapseUser(user, amount)
        else throw("Strict value is not equal to itself.")
    }



@Callable(i)
func lpValue (aTokenAmount) = $Tuple2(nil, aTokenToAsset(aTokenAmount))


@Verifier(tx)
func verify () = {
    let BASE = 1000
    let quorumRatio = valueOrErrorMessage(getInteger(configAddress, "proposal_quorum_ratio"), "proposalQuorumRatio is not defined")
    let passedRatio = valueOrErrorMessage(getInteger(configAddress, "proposal_passed_ratio"), "proposalThresholdRatio is not defined")
    let gViresContract = valueOrErrorMessage(addressFromString(valueOrErrorMessage(getString(configAddress, "dividends_contract"), "no dividends_contract")), "invalid dividends_contract")
    let votingContract = valueOrErrorMessage(addressFromString(valueOrErrorMessage(getString(configAddress, "voting_contract"), "no voting_contract")), "invalid voting_contract")
    let id = toBase58String(tx.id)
    let votesYes = valueOrElse(getInteger(votingContract, ("proposal_yes_" + id)), 0)
    let votesNo = valueOrElse(getInteger(votingContract, ("proposal_no_" + id)), 0)
    let proposalHeight = valueOrErrorMessage(getInteger(votingContract, ("proposal_height_" + id)), "proposal not registered")
    let applyStart = valueOrElse(getInteger(votingContract, ("proposal_applystart_" + id)), 0)
    let applyEnd = valueOrElse(getInteger(votingContract, ("proposal_applyend_" + id)), 0)
    let totalGVires = valueOrElse(getInteger(votingContract, ("proposal_gvires_" + id)), 0)
    let enabled = valueOrElse(getBoolean(configAddress, "op_governance_apply_tx_paused"), false)
    let voteYes = votesYes
    let voteNo = votesNo
    let totalVotes = (voteYes + voteNo)
    let hasQuorum = (((totalVotes * BASE) / totalGVires) >= quorumRatio)
    let hasPassed = (((voteYes * BASE) / totalVotes) >= passedRatio)
    let tooEarly = (applyStart >= HEIGHT)
    let tooLate = (HEIGHT >= applyEnd)
    let timeDebug = ((((((("proposalHeight = " + toString(proposalHeight)) + ", applyStart = ") + toString(applyStart)) + ", applyEnd = ") + toString(applyEnd)) + ", HEIGHT = ") + toString(HEIGHT))
    let byVoting = if (!(enabled))
        then throw("tx application throw governance not enabled")
        else if (tooEarly)
            then throw(("proposal can't be executed as it's too early: " + timeDebug))
            else if (tooLate)
                then throw(("proposal can't be executed as it's too late:" + timeDebug))
                else if (!(hasQuorum))
                    then throw((((((("no quorum: " + "totalVotes: ") + toString(totalVotes)) + ", totalGVires: ") + toString(totalGVires)) + ", quorumRatio: ") + toString(quorumRatio)))
                    else if (!(hasPassed))
                        then throw((((((("no threshold achieved: " + "voteYes: ") + toString(voteYes)) + ", voteNo: ") + toString(voteNo)) + ", passedRatio: ") + toString(passedRatio)))
                        else true
    if (sigVerify(tx.bodyBytes, tx.proofs[0], tx.senderPublicKey))
        then true
        else byVoting
    }
