func checkStreamRequirements (child,waveletsPerBlock,maxClaimable,isMaxClaimableUSD) = if ((0 >= waveletsPerBlock))
    then unit
    else {
        let effectiveMax = if ((maxClaimable > 0))
            then maxClaimable
            else if ((maxClaimable == 0))
                then valueOrElse(getInteger(maxClaimableStore(child)), 0)
                else 0
        if ((0 >= effectiveMax))
            then unit
            else assertChildDaoStarted(child)
        }


func availableNow (to) = {
    let perBlock = valueOrElse(getInteger(waveletsPerBlockStore(to)), 0)
    let total = if ((perBlock == 0))
        then 0
        else {
            let firstBlock = value(getInteger(startHeightStore(to)))
            let blocks = (HEIGHT - firstBlock)
            (perBlock * blocks)
            }
    let claimedWAVES = valueOrElse(getInteger(claimedStore(to)), 0)
    let restWaves = (total - claimedWAVES)
    let maxClaimable = getInteger(maxClaimableStore(to))
    if (isDefined(maxClaimable))
        then {
            let maxClaimableValue = value(maxClaimable)
            if (valueOrElse(getBoolean(maxClaimableIsUSDStore(to)), false))
                then {
                    let claimedUSD = valueOrElse(getInteger(claimedUSDStore(to)), wavesToUSD(claimedWAVES))
                    let restUSD = (maxClaimableValue - claimedUSD)
                    let restUsdInWaves = usdToWaves(restUSD)
                    min([restUsdInWaves, restWaves])
                    }
                else {
                    let grant = min([total, maxClaimableValue])
                    (grant - claimedWAVES)
                    }
            }
        else restWaves
    }

func setScheduleInternal (to,waveletsPerBlock,maxClaimable,isMaxClaimableUSD,sum) = {
    let adr = valueOrErrorMessage(addressFromString(to), ("invalid address: " + to))
    if ((adr == adr))
        then {
            let perBlockStore = waveletsPerBlockStore(to)
            let oldPerBlock = valueOrElse(getInteger(perBlockStore), 0)
            let requirements = checkStreamRequirements(to, waveletsPerBlock, maxClaimable, isMaxClaimableUSD)
            if ((requirements == requirements))
                then if (if (isMaxClaimableUSD)
                    then !(usdLimitedStreamsAllowed)
                    else false)
                    then throw("USD-limited streams are temporarily disabled. Use WAVES-denominated limits instead.")
                    else if ((0 > waveletsPerBlock))
                        then throw("waveletsPerBlock should be >= 0")
                        else if (if (isMaxClaimableUSD)
                            then (10000 > waveletsPerBlock)
                            else false)
                            then throw("10000 wavelets is a minimun to use within USD limits")
                            else if (if (isMaxClaimableUSD)
                                then (1000000 > maxClaimable)
                                else false)
                                then throw("USD limits should starts from 1000000 usdtlets")
                                else {
                                    let newSum = ((sum - oldPerBlock) + waveletsPerBlock)
                                    if ((newSum > maximum))
                                        then throw("maximum exceeded")
                                        else {
                                            let maxClaimableStore1 = maxClaimableStore(to)
                                            let maxClaimableIsUSDStore1 = maxClaimableIsUSDStore(to)
                                            let maxClaimableEntry = if ((maxClaimable == 0))
                                                then nil
                                                else if ((0 > maxClaimable))
                                                    then [DeleteEntry(maxClaimableStore1), DeleteEntry(maxClaimableIsUSDStore1)]
                                                    else [IntegerEntry(maxClaimableStore1, maxClaimable), BooleanEntry(maxClaimableIsUSDStore1, isMaxClaimableUSD)]
                                            let availableNowWaves = availableNow(to)
                                            let updatedStreams = if ((waveletsPerBlock > 0))
                                                then addStream(streams, to)
                                                else removeStream(streams, to)
                                            if (if ((updatedStreams != ""))
                                                then (size(split(updatedStreams, "|")) > MAX_STREAMS)
                                                else false)
                                                then throw((("max " + toString(MAX_STREAMS)) + " streams, can't add more"))
                                                else (([IntegerEntry(perBlockStore, waveletsPerBlock), IntegerEntry(startHeightStore(to), HEIGHT), IntegerEntry(claimedStore(to), -(availableNowWaves)), IntegerEntry(claimedUSDStore(to), -(wavesToUSD(availableNowWaves))), IntegerEntry(sumStore, newSum)] ++ maxClaimableEntry) ++ [StringEntry(streamsStore, updatedStreams)])
                                            }
                                    }
                else throw("Strict value is not equal to itself.")
            }
        else throw("Strict value is not equal to itself.")
    }


func leaseTarget (node) = {
    let votes = valueOrElse(getInteger(this, leaseNodeVotesSnapStore(node)), 0)
    if (if (if ((0 >= votes))
        then true
        else (0 >= leasePool))
        then true
        else (0 >= leaseTotalVotesSnap))
        then 0
        else if (!(isNodeSharing(node)))
            then 0
            else fraction(leasePool, votes, leaseTotalVotesSnap)
    }


func replaceLease (node,newAmount) = {
    let cancel =     match getBinary(this, leaseIdStore(node)) {
        case id: ByteVector => 
[LeaseCancel(id)]
        case _ => 
            nil
    }
    if ((newAmount > 0))
        then {
            let newLease = Lease(valueOrErrorMessage(addressFromString(node), ("invalid node address: " + node)), newAmount)
            (cancel ++ [newLease, BinaryEntry(leaseIdStore(node), calculateLeaseId(newLease)), IntegerEntry(leaseNodeAmountStore(node), newAmount)])
            }
        else (cancel ++ [DeleteEntry(leaseIdStore(node)), DeleteEntry(leaseNodeAmountStore(node))])
    }

        else false)
        then -1
        else 0

@Callable(i)
func Claim () = {
    let toAdr = i.caller
    let to = toString(toAdr)
    let a = valueOrErrorMessage(getInteger(waveletsPerBlockStore(to)), "address not found")
    if ((a == a))
        then {
            let eligibility = if (checkLockOnClaim)
                then {
                    let perBlock = valueOrElse(getInteger(waveletsPerBlockStore(to)), 0)
                    let maxCl = valueOrElse(getInteger(maxClaimableStore(to)), 0)
                    let isUSD = valueOrElse(getBoolean(maxClaimableIsUSDStore(to)), false)
                    checkStreamRequirements(to, perBlock, maxCl, isUSD)
                    }
                else unit
            if ((eligibility == eligibility))
                then {
                    let amount = availableNow(to)
                    let payment = min([wavesBalance(this).available, amount])
                    if ((0 >= payment))
                        then throw("nothing to claim")
                        else {
                            let claimedStore1 = claimedStore(to)
                            let claimed = valueOrElse(getInteger(claimedStore1), 0)
                            let claimedUSDStore1 = claimedUSDStore(to)
                            let claimedUSD = valueOrElse(getInteger(claimedUSDStore1), wavesToUSD(claimed))
                            let paymentUSD = wavesToUSD(payment)
                            if (if (valueOrElse(getBoolean(maxClaimableIsUSDStore(to)), false))
                                then (30 > paymentUSD)
                                else false)
                                then throw("usd amount too small")
                                else [ScriptTransfer(toAdr, payment, unit), IntegerEntry(claimedStore1, (claimed + payment)), IntegerEntry(claimedUSDStore1, (claimedUSD + paymentUSD))]
                            }
                    }
                else throw("Strict value is not equal to itself.")
            }
        else throw("Strict value is not equal to itself.")
    }
