func callerEnabled (address) = (indexOf(valueOrElse(getString("setup_shutdownCallers"), ""), toString(address)) != unit)


@Callable(i)
func addCaller (address) = if ((i.caller == this))
    then [StringEntry("setup_shutdownCallers", ((valueOrElse(getString("setup_shutdownCallers"), "") + address) + ","))]
    else throw("self invoke only")



@Callable(i)
func stopOne (addressStr,domain,bool) = if (callerEnabled(i.caller))
    then if (if ((addressStr == "3PEpv9hRFWEEBU22WRnLsw1bH4YGtcU728o"))
        then true
        else (domain == "pools"))
        then {
            let inv = invoke(Address(base58'3PEpv9hRFWEEBU22WRnLsw1bH4YGtcU728o'), "setStop", [bool], nil)
            if ((inv == inv))
                then nil
                else throw("Strict value is not equal to itself.")
            }
        else {
            let inv = invoke(addressFromStringValue(addressStr), "shutdown", [bool], nil)
            if ((inv == inv))
                then nil
                else throw("Strict value is not equal to itself.")
            }
    else throw("caller not enabled")
