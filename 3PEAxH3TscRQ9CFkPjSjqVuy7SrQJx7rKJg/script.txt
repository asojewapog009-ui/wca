@Callable(i)
func b () = if (if ((i.caller != executor))
    then (i.caller != this)
    else false)
    then throw("failed")
    else {
        let data = getStringValue(viresBtc, "reserveGlobalData")
        let dataArray = split(data, "|")
        let storedTotalDeposit = parseIntValue(dataArray[1])
        let storedTotalReserve = parseIntValue(dataArray[3])
        let storedTotalDebt = parseIntValue(dataArray[2])
        let maxWithdraw = ((storedTotalDeposit + storedTotalReserve) - storedTotalDebt)
        let balanceRes = invoke(viresBtc, "userBalance", [toString(this)], nil)
        if ((balanceRes == balanceRes))
            then             match balanceRes {
                case r: (Int, Int, Int, Int, Int, Boolean) => 
                    let withdrawBalance = if ((maxWithdraw > r._2))
                        then r._2
                        else maxWithdraw
                    if ((0 >= withdrawBalance))
                        then throw("zero")
                        else {
                            let res = invoke(viresMain, "withdraw2", [viresBtcStr, withdrawBalance], nil)
                            if ((res == res))
                                then nil
                                else throw("Strict value is not equal to itself.")
                            }
                case _ => 
                    throw("wrong type")
            }
            else throw("Strict value is not equal to itself.")
        }
