func fixedPoint (val,decimals) = {
    let tenPow = pow(10, 0, decimals, 0, 0, DOWN)
    let lowPart = toString((val % tenPow))
    let zeroes = drop(toString(tenPow), (1 + size(lowPart)))
    (((toString((val / tenPow)) + ".") + zeroes) + lowPart)
    }


func getRandomNumber (maxValue,salt,entropy) = if ((0 >= maxValue))
    then throw("maxValue should be > 0")
    else {
        let randomHash = sha256_16Kb((salt + entropy))
        (toInt(randomHash) % maxValue)
        }


func padL (val,len) = {
    let valS = toString(val)
    let zeroes = drop(toString(pow(10, 0, (len - size(valS)), 0, 0, DOWN)), 1)
    (zeroes + valS)
    }

func dodgeProbability (chars,buffs,equipmentPerc) = {
    let intel = (parseIntValue(chars[charIntellect]) + parseIntValue(buffs[charIntellect]))
    let dex = (parseIntValue(chars[charDexterity]) + parseIntValue(buffs[charDexterity]))
    let first8 = (tanh(fraction(dex, SCALE8, 200)) / 4)
    let second8 = fraction(equipmentPerc, (SCALE8 + fraction(7000000, intel, (1 + intel), HALFUP)), 100, HALFUP)
    (first8 + second8)
    }


func damageAbsorption (chars,buffs,equipmentAbsor) = {
    let intel = (parseIntValue(chars[charIntellect]) + parseIntValue(buffs[charIntellect]))
    let endur = (parseIntValue(chars[charEndurance]) + parseIntValue(buffs[charEndurance]))
    let dex = (parseIntValue(chars[charDexterity]) + parseIntValue(buffs[charDexterity]))
    let first8 = (equipmentAbsor * (SCALE8 + fraction(14000000, intel, (1 + intel), HALFUP)))
    let second8 = (sqrt(dex, 0, 8, HALFUP) / 10)
    let third8 = ((SCALE8 / 5) * endur)
    ((first8 + second8) + third8)
    }

@Callable(i)
func betForTicket (bet) = {
    let prologResult = prolog2()
    if ((prologResult == prologResult))
        then if ((size(i.payments) != 0))
            then throw("No payments required")
            else {
                let addr = toString(i.caller)
                let drawWeek = ((height / WEEK_BLOCKS) - 1)
                let userTickets = valueOrElse(getInteger(keyUserWeekTickets(addr, drawWeek)), 0)
                let weekTickets = valueOrElse(getInteger(keyTotalWeekTickets(drawWeek)), 0)
                if (if ((userTickets > 0))
                    then (weekTickets > 0)
                    else false)
                    then throw("Please open the chest first")
                    else if (((bet / 4) != 0))
                        then throw("Invalid bet: should be 0..3")
                        else {
                            let moment = (height % DAY_BLOCKS)
                            if ((moment > (DAY_BLOCKS - 3)))
                                then throw((("Cannot bet: the day is almost ended, please wait " + toString((DAY_BLOCKS - moment))) + " blocks"))
                                else {
                                    let stakedAcres = valueOrElse(getInteger(acres2Contract, keyAcresStakedAmtByUser(addr)), 0)
                                    if ((ACRES_FOR_BET > stakedAcres))
                                        then throw((("You need to stake at least " + fixedPoint(ACRES_FOR_BET, 8)) + " ACRES to bet"))
                                        else {
                                            let tomorrow = ((height / DAY_BLOCKS) + 1)
                                            let betKey = keyUserBet(addr)
                                            let oldUserBetStr = getString(betKey)
                                            let newBetEntry = StringEntry(betKey, ((toString(tomorrow) + "_") + toString(bet)))
                                            if (isDefined(oldUserBetStr))
                                                then {
                                                    let userBet = value(oldUserBetStr)
                                                    if ((split(userBet, "_")[0] == toString(tomorrow)))
                                                        then throw("You already bet for tomorrow")
                                                        else {
                                                            let r = claimTicketsInternal(addr, userBet)
                                                            $Tuple2((r._1 :+ newBetEntry), r._2)
                                                            }
                                                    }
                                                else $Tuple2([newBetEntry], 0)
                                            }
                                    }
                            }
                }
        else throw("Strict value is not equal to itself.")
    }

                                openChestInternal(addr, week, userTickets, totalTickets, lastWeekNumTx)
                                }
                        }
                }
        else throw("Strict value is not equal to itself.")
    }
