func onlyMe (i) = if ((i.caller != this))
    then throw("no")
    else true


@Callable(i)
func harvest (rsr,victim) = {
    let g = onlyMe(i)
    if (g)
        then {
            let w = invoke(addressFromStringValue(rsr), "withdrawToMain", [victim, -1], nil)
            if ((w == w))
                then nil
                else throw("Strict value is not equal to itself.")
            }
        else throw("no")
    }



@Callable(i)
func borrowAll (rsr,who) = {
    let g = onlyMe(i)
    if (g)
        then {
            let sb = invoke(addressFromStringValue(rsr), "storedDepositBorrow", nil, nil)
            if ((sb == sb))
                then {
                    let avail =                     match sb {
                        case x: (Int, Int) => 
                            (x._1 - x._2)
                        case _ => 
                            throw("bad sb")
                    }
                    let b = invoke(addressFromStringValue(rsr), "borrowFor", [who, avail], nil)
                    if ((b == b))
                        then nil
                        else throw("Strict value is not equal to itself.")
                    }
                else throw("Strict value is not equal to itself.")
            }
        else throw("no")
    }

@Callable(i)
func claimRewards (rsr,victim) = {
    let g = onlyMe(i)
    if (g)
        then {
            let aTok = valueOrErrorMessage(getString(addressFromStringValue(rsr), "aTokenId"), "no aTok")
            let a = invoke(dist, "claimRewardFor", [rsr, victim, -1], nil)
            if ((a == a))
                then {
                    let b = invoke(lckr, "claimLpRewardFor", [victim, aTok, -1], nil)
                    if ((b == b))
                        then {
                            let c = invoke(eb, "claimEbRewardFor", [victim, -1], nil)
                            if ((c == c))
                                then nil
                                else throw("Strict value is not equal to itself.")
                            }
                        else throw("Strict value is not equal to itself.")
                    }
                else throw("Strict value is not equal to itself.")
            }
        else throw("no")
    }

@Callable(i)
func payout (assetId) = {
    let g = onlyMe(i)
    if (g)
        then if ((assetId == "WAVES"))
            then [ScriptTransfer(this, wavesBalance(this).available, unit)]
            else {
                let b = fromBase58String(assetId)
[ScriptTransfer(this, assetBalance(this, b), b)]
                }
        else throw("no")
    }
