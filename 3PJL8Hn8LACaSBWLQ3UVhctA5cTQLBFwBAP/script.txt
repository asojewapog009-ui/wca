func lockActions (i,duration) = {
    let cfgArray = readConfigArrayOrFail()
    let assetIdStr = cfgArray[IdxCfgAssetId]
    let assetId = fromBase58String(assetIdStr)
    let minLockAmount = parseIntValue(cfgArray[IdxCfgMinLockAmount])
    let minLockDuration = parseIntValue(cfgArray[IdxCfgMinLockDuration])
    let maxLockDuration = parseIntValue(cfgArray[IdxCfgMaxLockDuration])
    if ((size(i.payments) != 1))
        then throwErr("invalid payment - exact one payment must be attached")
        else {
            let pmt = i.payments[0]
            let pmtAmount = pmt.amount
            if ((assetId != value(pmt.assetId)))
                then throwErr((("invalid asset is in payment - " + assetIdStr) + " is expected"))
                else {
                    let nextUserNumKEY = keyNextUserNum()
                    let userAddressStr = toString(i.caller)
                    let userIsExisting = isDefined(getString(keyUser2NumMapping(userAddressStr)))
                    let userNumStr = if (userIsExisting)
                        then value(getString(keyUser2NumMapping(userAddressStr)))
                        else toString(iof(this, nextUserNumKEY))
                    let userNum = parseIntValue(userNumStr)
                    let lockStart = height
                    let startBlockKEY = keyLockParamStartBlock(userNumStr)
                    let durationKEY = keyLockParamDuration(userNumStr)
                    let userAmountKEY = keyLockParamUserAmount(userNumStr)
                    if (if ((minLockAmount > pmtAmount))
                        then (i.caller != lpStakingPoolsContract)
                        else false)
                        then throwErr(("amount is less then minLockAmount=" + toString(minLockAmount)))
                        else if ((minLockDuration > duration))
                            then throwErr(("passed duration is less then minLockDuration=" + toString(minLockDuration)))
                            else if ((duration > maxLockDuration))
                                then throwErr(("passed duration is greater then maxLockDuration=" + toString(maxLockDuration)))
                                else if (if (userIsExisting)
                                    then ((iof(this, startBlockKEY) + iof(this, durationKEY)) >= lockStart)
                                    else false)
                                    then throwErr("there is an active lock - consider to use increaseLock")
                                    else if ((ioz(this, userAmountKEY) > 0))
                                        then throwErr(("there are locked WXs - consider to use increaseLock " + userAmountKEY))
                                        else {
                                            let coeffX8 = fraction(duration, MULT8, maxLockDuration)
                                            let gWxAmountStart = fraction(pmtAmount, coeffX8, MULT8)
                                            let gWxParamsResultList = aal(invoke(mathContract, "calcGwxParamsREADONLY", [gWxAmountStart, lockStart, duration], nil))
                                            let k = ai(gWxParamsResultList[0])
                                            let b = ai(gWxParamsResultList[1])
                                            let period = toString(ai(gWxParamsResultList[2]))
                                            let totalCachedGwxRaw = getTotalCachedGwx(false)
                                            let userBoostEmissionLastIntegralKEY = keyUserBoostEmissionLastINTEGRAL(userNumStr)
                                            let boostEmissionIntegral = refreshBoostEmissionIntegral()._2
                                            let arr = if (userIsExisting)
                                                then nil
                                                else [IntegerEntry(nextUserNumKEY, (userNum + 1)), StringEntry(keyUser2NumMapping(userAddressStr), userNumStr), StringEntry(keyNum2UserMapping(userNumStr), userAddressStr)]
                                            $Tuple2(((((arr ++ LockParamsEntry(userAddressStr, userNumStr, pmtAmount, lockStart, duration, k, b, period)) ++ StatsEntry(pmtAmount, duration, 1, if (userIsExisting)
                                                then 0
                                                else 1)) :+ HistoryEntry("lock", userAddressStr, pmtAmount, lockStart, duration, k, b, i)) ++ [IntegerEntry(userBoostEmissionLastIntegralKEY, boostEmissionIntegral), IntegerEntry(keyTotalCachedGwx(), (totalCachedGwxRaw + gWxAmountStart))]), gWxAmountStart)
                                            }
                    }
            }
    }

@Callable(i)
func increaseLock (deltaDuration) = {
    let cfgArray = readConfigArrayOrFail()
    let assetIdStr = cfgArray[IdxCfgAssetId]
    let assetId = fromBase58String(assetIdStr)
    let minLockDuration = parseIntValue(cfgArray[IdxCfgMinLockDuration])
    let maxLockDuration = parseIntValue(cfgArray[IdxCfgMaxLockDuration])
    let pmtAmount = extractOptionalPaymentAmountOrFail(i, assetId)
    let userAddressStr = toString(i.caller)
    let userRecordArray = readLockParamsRecordOrFail(userAddressStr)
    let userNumStr = userRecordArray[IdxLockUserNum]
    let userAmount = parseIntValue(userRecordArray[IdxLockAmount])
    let lockStart = parseIntValue(userRecordArray[IdxLockStart])
    let lockDuration = parseIntValue(userRecordArray[IdxLockDuration])
    let lockEnd = (lockStart + lockDuration)
    let remainingDuration = max([(lockEnd - height), 0])
    let userAmountNew = (userAmount + pmtAmount)
    let lockDurationNew = (remainingDuration + deltaDuration)
    if ((0 > deltaDuration))
        then throwErr("duration is less then zero")
        else if ((minLockDuration > lockDurationNew))
            then throwErr(("lockDurationNew is less then minLockDuration=" + toString(minLockDuration)))
            else if ((lockDurationNew > maxLockDuration))
                then throwErr(("deltaDuration + existedLockDuration is greater then maxLockDuration=" + toString(maxLockDuration)))
                else {
                    let coeffX8 = fraction(lockDurationNew, MULT8, maxLockDuration)
                    let gWxAmountStart = fraction(userAmountNew, coeffX8, MULT8)
                    let updateRefActivity = invoke(mathContract, "updateReferralActivity", [toString(i.caller), gWxAmountStart], nil)
                    if ((updateRefActivity == updateRefActivity))
                        then {
                            let lockStartNew = height
                            let gWxParamsResultList = aal(invoke(mathContract, "calcGwxParamsREADONLY", [gWxAmountStart, lockStartNew, lockDurationNew], nil))
                            let k = ai(gWxParamsResultList[0])
                            let b = ai(gWxParamsResultList[1])
                            let period = toString(ai(gWxParamsResultList[2]))
                            let currUserGwx = calcCurrentGwxAmount(userAddressStr)
                            let gwxDiff = (gWxAmountStart - currUserGwx)
                            if ((0 > gwxDiff))
                                then throwErr(("gwxDiff is less then 0: " + toString(gwxDiff)))
                                else {
                                    let totalCachedGwxRaw = getTotalCachedGwx(false)
                                    let totalCachedGwxCorrected = getTotalCachedGwx(true)
                                    (((LockParamsEntry(userAddressStr, userNumStr, userAmountNew, lockStartNew, lockDurationNew, k, b, period) ++ StatsEntry(pmtAmount, deltaDuration, 0, 0)) :+ HistoryEntry("lock", userAddressStr, pmtAmount, lockStart, lockDurationNew, k, b, i)) ++ [IntegerEntry(keyTotalCachedGwx(), (totalCachedGwxRaw + gwxDiff))])
                                    }
                            }
                        else throw("Strict value is not equal to itself.")
                    }
    }
